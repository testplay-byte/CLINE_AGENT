   116→function getHeredocDelimiter(command: string): string | undefined {
   117→        const match = command.match(
   118→                /(?<![<])<<-?\s*(?:"([^"]+)"|'([^']+)'|([A-Za-z0-9_./-]+))/,
   119→        );
   120→        return match?.[1] ?? match?.[2] ?? match?.[3];
   121→}
   122→
   123→function coalesceSplitHeredocCommands(commands: string[]): string[] {
   124→        const coalesced: string[] = [];
   125→        for (let index = 0; index < commands.length; index += 1) {
   126→                const command = commands[index];
   127→                const delimiter = getHeredocDelimiter(command);
   128→                if (!delimiter) {
   129→                        coalesced.push(command);
   130→                        continue;
   131→                }
   132→
   133→                const endIndex = commands.findIndex(
   134→                        (nextCommand, nextIndex) =>
   135→                                nextIndex > index && nextCommand.trim() === delimiter,
   136→                );
   137→                if (endIndex === -1) {
   138→                        coalesced.push(command);
   139→                        continue;
   140→                }
   141→
   142→                const parts = [command];
   143→                while (index < endIndex) {
   144→                        index += 1;
   145→                        const nextCommand = commands[index];
   146→                        parts.push(nextCommand);
   147→                }
   148→                coalesced.push(parts.join("\n"));
   149→        }
   150→        return coalesced;
   151→}
   152→
   153→function coalesceAdjacentStringHeredocs(
   154→        commands: Array<string | StructuredCommandInput>,
   155→): Array<string | StructuredCommandInput> {
   156→        const coalesced: Array<string | StructuredCommandInput> = [];
   157→        let stringRun: string[] = [];
   158→
   159→        const flushStringRun = () => {
   160→                if (stringRun.length > 0) {
   161→                        coalesced.push(...coalesceSplitHeredocCommands(stringRun));
   162→                        stringRun = [];
   163→                }
   164→        };
   165→
   166→        for (const command of commands) {
   167→                if (typeof command === "string") {
   168→                        stringRun.push(command);
   169→                        continue;
   170→                }
   171→
   172→                flushStringRun();
   173→                coalesced.push(command);
   174→        }
   175→
   176→        flushStringRun();
   177→        return coalesced;
   178→}

   180→async function executeShellCommands(
   181→        commands: Array<string | StructuredCommandInput>,
   182→        options: {
   183→                executor: ShellExecutor;
   184→                cwd: string;
   185→                context: AgentToolContext;
   186→                timeoutMs: number;
   187→                timeoutSource: "default_setting" | "configured_setting";
   188→                telemetry?: ITelemetryService;
   189→        },
   190→): Promise<ToolOperationResult[]> {
   191→        const { executor, cwd, context, timeoutMs, timeoutSource, telemetry } =
   192→                options;
   193→
   194→        return Promise.all(
   195→                commands.map(async (command): Promise<ToolOperationResult> => {
   196→                        const startedAt = Date.now();
   197→                        const query = formatRunCommandQueryPreview(command);
   198→                        try {
   199→                                const output = await withTimeout(
   200→                                        executor(command, cwd, context),
   201→                                        timeoutMs,
   202→                                        `Command timed out after ${timeoutMs}ms`,
   203→                                );
   204→                                return {
   205→                                        query,
   206→                                        result: output,
   207→                                        success: true,
   208→                                };
   209→                        } catch (error) {
   210→                                if (error instanceof TimeoutError) {
   211→                                        captureRunCommandsTimeoutFromContext(telemetry, context, {
   212→                                                effectiveTimeoutMs: error.timeoutMs,
   213→                                                timeoutSource,
   214→                                                commandCount: commands.length,
   215→                                                durationMs: Date.now() - startedAt,
   216→                                        });
   217→                                }
   218→                                if (error instanceof CommandExitError) {
   219→                                        return {
   220→                                                query,
   221→                                                result: error.output,
   222→                                                error: error.message,
   223→                                                success: false,
   224→                                        };
   225→                                }
   226→                                const msg = formatError(error);
   227→                                return {
   228→                                        query,
   229→                                        result: "",
   230→                                        error: `Command failed: ${msg}`,
   231→                                        success: false,
   232→                                };
   233→                        }
   234→                }),
   235→        );
   236→}

   242→/**
   243→ * Create the read_files tool
   244→ *
   245→ * Reads the content of one or more files from the filesystem.
   246→ */
   247→export function createReadFilesTool(
   248→        executor: FileReadExecutor,
   249→        config: Pick<DefaultToolsConfig, "fileReadTimeoutMs"> = {},
   250→): AgentTool<ReadFilesInput, ToolOperationResult[]> {
   251→        const timeoutMs = config.fileReadTimeoutMs ?? 10000;
   252→
   253→        return createTool<ReadFilesInput, ToolOperationResult[]>({
   254→                name: "read_files",
   255→                description:
   256→                        "Read the content of text or image files at the provided absolute paths, or return only an inclusive one-based line range when start_line/end_line are provided on the same file entry as its path. " +
   257→                        "When you already know multiple files you need, read them together in one call, and call this tool in the same response as other independent tool calls. " +
   258→                        `Each read returns at most ${MAX_READ_LINES} lines / ~${Math.round(MAX_READ_OUTPUT_CHARS / 1024)}k characters; longer files report their total line count, page through them with start_line/end_line on that file's entry. ` +
   259→                        "Binary files that are not image and large files are not supported. " +
   260→                        "Returns file contents or error messages for each path. ",
   261→                inputSchema: zodToJsonSchema(ReadFilesInputSchema),
   262→                timeoutMs: timeoutMs * 2, // Account for multiple files
   263→                retryable: true,
   264→                maxRetries: 1,
   265→                execute: async (input, context) => {
   266→                        const validate = validateWithZod(
   267→                                ReadFilesInputUnionSchema,
   268→                                coalesceOrphanReadRanges(input),
   269→                        );
   270→                        let requests: ReadFileRequest[];
   271→                        if (typeof validate === "string") {
   272→                                requests = [{ path: validate }];
   273→                        } else if (Array.isArray(validate)) {
   274→                                requests = validate.map((value) =>
   275→                                        typeof value === "string" ? { path: value } : value,
   276→                                );
   277→                        } else if ("files" in validate) {
   278→                                const files = Array.isArray(validate.files)
   279→                                        ? validate.files
   280→                                        : [validate.files];
   281→                                requests = files.map((file) =>
   282→                                        typeof file === "string" ? { path: file } : file,
   283→                                );
   284→                        } else if ("file_paths" in validate) {
   285→                                const filePaths = Array.isArray(validate.file_paths)
   286→                                        ? validate.file_paths
   287→                                        : [validate.file_paths];
   288→                                requests = filePaths.map((path) => ({ path }));
   289→                        } else if ("paths" in validate) {
   290→                                const paths = Array.isArray(validate.paths)
   291→                                        ? validate.paths
   292→                                        : [validate.paths];
   293→                                requests = paths.map((path) =>
   294→                                        typeof path === "string" ? { path } : path,
   295→                                );
   296→                        } else {
   297→                                requests = [validate];
   298→                        }
   299→
   300→                        return Promise.all(
   301→                                requests.map(async (request): Promise<ToolOperationResult> => {
   302→                                        const rangeError = getReadFileRangeError(request);
   303→                                        if (rangeError) {
   304→                                                return {
   305→                                                        query: formatReadFileQuery(request),
   306→                                                        result: "",
   307→                                                        error: `Invalid file range: ${rangeError}`,
   308→                                                        success: false,
   309→                                                };
   310→                                        }
   311→
   312→                                        try {
   313→                                                const content = await withTimeout(
   314→                                                        executor(request, context),
   315→                                                        timeoutMs,
   316→                                                        `File read timed out after ${timeoutMs}ms`,
   317→                                                );
   318→                                                return {
   319→                                                        query: formatReadFileQuery(request),
   320→                                                        result: content,
   321→                                                        success: true,
   322→                                                };
   323→                                        } catch (error) {
   324→                                                const msg = formatError(error);
   325→                                                return {
   326→                                                        query: formatReadFileQuery(request),
   327→                                                        result: "",
   328→                                                        error: `Error reading file: ${msg}`,
   329→                                                        success: false,
   330→                                                };
   331→                                        }
   332→                                }),
   333→                        );
   334→                },
   335→        });
   336→}

   404→/**
   405→ * Build the run_commands tool description for the shell that will actually
   406→ * execute the commands. The shell kind decides the syntax guidance (quoting,
   407→ * sequencing, heredocs), and isWindows adds environment context for POSIX
   408→ * shells running on a Windows host (e.g. Git Bash).
   409→ */
   410→export function buildRunCommandsDescription(
   411→        shellKind: ShellKind,
   412→        isWindows: boolean,
   413→): string {
   414→        if (shellKind === "powershell" || shellKind === "cmd") {
   415→                const shellName = shellKind === "powershell" ? "PowerShell" : "cmd.exe";
   416→                const sequencingOperator = shellKind === "powershell" ? "';'" : "'&&'";
   417→                return (
   418→                        "Run non-interactive shell commands from the root of the workspace in Windows environment. " +
   419→                        RUN_COMMANDS_SHARED_INSTRUCTIONS +
   420→                        `Output beyond ~${Math.round(MAX_COMMAND_OUTPUT_CHARS / 1000)}k characters is middle-truncated (start and end preserved); filter output when you need specific sections. ` +
   421→                        `Commands run through ${shellName}; quote paths and arguments for ${shellName} and use ${sequencingOperator} to sequence commands. ` +
   422→                        "Include multiple commands in the same call when they are independent and safe to run concurrently. When independent reads, searches, or edits are also needed, call those tools in the same response."
   423→                );
   424→        }
   425→
   426→        const environmentNote =
   427→                shellKind === "wsl"
   428→                        ? "Commands run through bash in WSL (wsl.exe); the Windows working directory is mounted under /mnt/<drive>. "
   429→                        : isWindows
   430→                                ? "Commands run through a POSIX (bash-compatible) shell on Windows. "
   431→                                : "";
   432→        return (
   433→                "Run non-interactive shell commands from the root of the workspace. " +
   434→                RUN_COMMANDS_SHARED_INSTRUCTIONS +
   435→                environmentNote +
   436→                "Commands should be properly shell-escaped and targeted to avoid error or timeout. Include multiple commands in the same call when they are independent complete shell commands and safe to run concurrently; multiline scripts and heredocs must be a single command string. When independent reads, searches, or edits are also needed, call those tools in the same response. " +
   437→                `Output beyond ~${Math.round(MAX_COMMAND_OUTPUT_CHARS / 1000)}k characters is middle-truncated (start and end preserved); pipe through grep/head/tail when you need specific sections of large output. ` +
   438→                "For long-running commands, run them in background and redirect output to a tmp file that you can read from later."
   439→        );
   440→}

   442→/**
   443→ * Create the run_commands shell tool for the current platform.
   444→ *
   445→ * This preserves the SDK's platform-specific prompting/schema choices while
   446→ * exposing a single generic shell-tool factory for host integrations. Pass
   447→ * config.shell (matching the executor's shell) so the syntax guidance in the
   448→ * tool description matches the shell that actually runs the commands.
   449→ *
   450→ * config.shell may be a provider function instead of a string. The runtime
   451→ * reads `description` when building each model request, so a provider is
   452→ * consulted at that boundary: a shell change made while the model is
   453→ * generating does not affect the request in flight, and the next request
   454→ * names the new shell. The provider must return the shell the executor will
   455→ * use for tool calls issued by that next request.
   456→ */
   457→export function createShellTool(
   458→        executor: ShellExecutor,
   459→        config: Pick<DefaultToolsConfig, "cwd" | "bashTimeoutMs" | "telemetry"> & {
   460→                shell?: string | (() => string);
   461→        } = {},
   462→): AgentTool<unknown, ToolOperationResult[]> {
   463→        const timeoutMs = config.bashTimeoutMs ?? 30000;
   464→        const timeoutSource =
   465→                config.bashTimeoutMs === undefined
   466→                        ? "default_setting"
   467→                        : "configured_setting";
   468→        const cwd = config.cwd ?? process.cwd();
   469→        const isWindows = process.platform === "win32";
   470→        const configShell = config.shell;
   471→        const resolveShell =
   472→                typeof configShell === "function"
   473→                        ? configShell
   474→                        : () => configShell ?? getDefaultShell(process.platform);
   475→        const describe = () =>
   476→                buildRunCommandsDescription(getShellKind(resolveShell()), isWindows);
   477→
   478→        const tool = createTool<unknown, ToolOperationResult[]>({
   479→                name: "run_commands",
   480→                description: describe(),
   481→                inputSchema: zodToJsonSchema(RunCommandsInputSchema),
   482→                timeoutMs: timeoutMs * 2,
   483→                retryable: false,
   484→                maxRetries: 0,
   485→                execute: async (input, context) => {
   486→                        const commands = coalesceAdjacentStringHeredocs(
   487→                                normalizeRunCommandsInput(input),
   488→                        );
   489→
   490→                        return executeShellCommands(commands, {
   491→                                executor,
   492→                                cwd,
   493→                                context,
   494→                                timeoutMs,
   495→                                timeoutSource,
   496→                                telemetry: config.telemetry,
   497→                        });
   498→                },
   499→        });
   500→
   501→        if (typeof configShell === "function") {
   502→                // The runtime rebuilds tool definitions from this property for every
   503→                // model request, so a getter re-derives the description at exactly the
   504→                // send-to-model boundary. AgentTool consumers only read `description`.
   505→                Object.defineProperty(tool, "description", {
   506→                        get: describe,
   507→                        enumerable: true,
   508→                });
   509→        }
   510→        return tool;
   511→}

   608→ *
   609→ * Applies the canonical apply_patch format to one or more files.
   610→ */
   611→export function createApplyPatchTool(
   612→        executor: ApplyPatchExecutor,
   613→        config: Pick<DefaultToolsConfig, "cwd" | "applyPatchTimeoutMs"> = {},
   614→): AgentTool<ApplyPatchInput, ToolOperationResult> {
   615→        const timeoutMs = config.applyPatchTimeoutMs ?? 30000;
   616→        const cwd = config.cwd ?? process.cwd();
   617→
   618→        return createTool<ApplyPatchInput, ToolOperationResult>({
   619→                name: "apply_patch",
   620→                description: APPLY_PATCH_TOOL_DESC,
   621→                inputSchema: zodToJsonSchema(ApplyPatchInputSchema),
   622→                timeoutMs,
   623→                retryable: false,
   624→                maxRetries: 0,
   625→                execute: async (input, context) => {
   626→                        const validate = validateWithZod(ApplyPatchInputUnionSchema, input);
   627→                        const patchInput =
   628→                                typeof validate === "string" ? validate : validate.input;
   629→
   630→                        try {
   631→                                const result = await withTimeout(
   632→                                        executor({ input: patchInput }, cwd, context),
   633→                                        timeoutMs,
   634→                                        `apply_patch timed out after ${timeoutMs}ms`,
   635→                                );
   636→
   637→                                return {
   638→                                        query: "apply_patch",
   639→                                        result,
   640→                                        success: true,
   641→                                };
   642→                        } catch (error) {
   643→                                const msg = formatError(error);
   644→                                return {
   645→                                        query: "apply_patch",
   646→                                        result: "",
   647→                                        error: `apply_patch failed: ${msg}`,
   648→                                        success: false,
   649→                                };
   650→                        }
   651→                },
   652→        });

   666→
   667→        return createTool<EditFileInput, ToolOperationResult>({

   669→                description:
   670→                        "An editor for controlled filesystem edits on the text file at the provided path. " +
   671→                        "Provide `insert_line` to insert `new_text` at a specific line number. " +
   672→                        "Otherwise, the tool replaces `old_text` with `new_text`, or creates the file with `new_text` if file does not exist. " +
   673→                        "Use this tool for making small, precise edits to existing files or creating new files over shell commands. If several edits to different files or non-overlapping regions are already known, emit multiple editor tool calls in the same response instead of serializing them across turns.",
   674→
   675→                inputSchema: zodToJsonSchema(EditFileInputSchema),
   676→                timeoutMs,
   677→                retryable: false, // Editing operations are stateful and should not auto-retry

   690→                                };
   691→                        }
   692→
   693→                        try {
   694→                                const result = await withTimeout(
   695→                                        executor(validatedInput, cwd, context),

   697→                                        `Editor operation timed out after ${timeoutMs}ms`,

   700→                                return {
   701→                                        query: `${operation}:${validatedInput.path}`,
   702→                                        result,
   703→                                        success: true,

   705→                        } catch (error) {

   708→                                        query: `${operation}:${validatedInput.path}`,
   709→                                        result: "",

   723→export function createSkillsTool(
   724→        executor: SkillsExecutorWithMetadata,

   726→): AgentTool<SkillsInput, string> {

   729→        const baseDescription =
   730→                "Execute a skill within the main conversation. " +
   731→                "When users ask you to perform tasks, check if any available skills match. " +
   732→                "When users reference a slash command, invoke it with this tool. " +
   733→                'Input: `skill` (required) and optional `args`. Example: `skill: "pdf"`, `skill: "commit", args: "-m \\"Fix bug\\""`, `skill: "review-pr", args: "123"`, `skill: "ms-office-suite:pdf"`. ' +
   734→                "When a skill matches the user's request, invoking this tool is a blocking requirement before any other response. " +
   735→                "Never mention a skill without invoking this tool.";
   736→
   737→        const tool = createTool<SkillsInput, string>({
   738→                name: "skills",
   739→                description: baseDescription,
   740→                inputSchema: zodToJsonSchema(SkillsInputSchema),

   746→                        return withTimeout(
   747→                                executor(
   748→                                        validatedInput.skill,
   749→                                        validatedInput.args || undefined,
   750→                                        context,
   751→                                ),
   752→                                timeoutMs,
   753→                                `Skills operation timed out after ${timeoutMs}ms`,
   754→                        );
   755→                },
   756→        });
   757→

   759→                get() {

   767→                },
   768→                enumerable: true,
   769→                configurable: true,
   770→        });
   771→
   772→        return tool;

   837→/**
   838→ * Create a set of default tools for an agent
   839→ *
   840→ * This function creates the default tools based on the provided configuration
   841→ * and executors. Only tools that are enabled AND have an executor provided
   842→ * will be included in the returned array.
   843→ *
   844→ * @example
   845→ * ```typescript
   846→ * import { Agent, createDefaultTools } from "@cline/core"
   847→ * import * as fs from "fs/promises"
   848→ * import { exec } from "child_process"
   849→ *
   850→ * const tools = createDefaultTools({
   851→ *   executors: {
   852→ *     readFile: async ({ path }) => fs.readFile(path, "utf-8"),
   853→ *     bash: async (cmd, cwd) => {
   854→ *       return new Promise((resolve, reject) => {
   855→ *         exec(cmd, { cwd }, (err, stdout, stderr) => {
   856→ *           if (err) reject(new Error(stderr || err.message))
   857→ *           else resolve(stdout)
   858→ *         })
   859→ *       })
   860→ *     },
   861→ *   },
   862→ *   enableReadFiles: true,
   863→ *   enableBash: true,
   864→ *   enableSearch: false, // Disabled
   865→ *   enableWebFetch: false, // Disabled
   866→ *   cwd: "/path/to/project",
   867→ * })
   868→ *
   869→ * const agent = new Agent({
   870→ *   // ... provider config
   871→ *   tools,
   872→ * })
   873→ * ```
   874→ */
   875→export function createDefaultTools(
   876→        options: CreateDefaultToolsOptions,
   877→): AgentTool[] {
   878→        const {
   879→                executors,
   880→                enableReadFiles = true,
   881→                enableSearch = true,
   882→                enableBash = true,
   883→                enableWebFetch = true,
   884→                enableApplyPatch = false,
   885→                enableEditor = true,
   886→                enableSkills = true,
   887→                enableAskQuestion = true,
   888→                enableSubmitAndExit = false,
   889→                ...config
   890→        } = options;
   891→
   892→        const tools: AgentTool<never, unknown>[] = [];
   893→
   894→        // Add read_files tool if enabled and executor provided
   895→        if (enableReadFiles && executors.readFile) {
   896→                tools.push(createReadFilesTool(executors.readFile, config));
   897→        }
   898→
   899→        // Add search_codebase tool if enabled and executor provided
   900→        if (enableSearch && executors.search) {
   901→                tools.push(createSearchTool(executors.search, config));
   902→        }
   903→
   904→        // Add run_commands tool if enabled and executor provided
   905→        if (enableBash && executors.bash) {
   906→                tools.push(createShellTool(executors.bash, config));
   907→        }
   908→
   909→        // Add fetch_web_content tool if enabled and executor provided
   910→        if (enableWebFetch && executors.webFetch) {
   911→                tools.push(createWebFetchTool(executors.webFetch, config));
   912→        }
   913→
   914→        // Add editor tool if enabled and executor provided,
   915→        // else check if apply_patch tool is enabled and executor provided
   916→        // NOTE: Do not enable two similar tools at the same time.
   917→        if (enableEditor && executors.editor) {
   918→                tools.push(createEditorTool(executors.editor, config));
   919→        } else if (enableApplyPatch && executors.applyPatch) {
   920→                tools.push(createApplyPatchTool(executors.applyPatch, config));
   921→        }
   922→
   923→        // Add skills tool if enabled and executor provided
   924→        if (enableSkills && executors.skills) {
   925→                tools.push(createSkillsTool(executors.skills, config));
   926→        }
   927→
   928→        const submitExecutor = enableSubmitAndExit ? executors.submit : undefined;
   929→
   930→        // Add ask_question tool if enabled and executor provided
   931→        if (enableAskQuestion && executors.askQuestion && !submitExecutor) {
   932→                tools.push(createAskQuestionTool(executors.askQuestion));
   933→        }
   934→
   935→        // Add submit_and_exit tool if enabled and executor provided
   936→        if (submitExecutor) {
   937→                tools.push(createSubmitAndExitTool(submitExecutor, config));
   938→        }
   939→
   940→        return tools as unknown as AgentTool[];
   941→}
