   633→                                                // higher seq, fence stale epochs, never let an out-of-order or duplicate

   639→                                                        // Stale/ignored — no change.
   640→                                                        return prevState
   641→                                                }
   642→                                                return { ...prevState, clineMessages: replicaRef.current.messages }
   643→                                        })

   646→                                }
