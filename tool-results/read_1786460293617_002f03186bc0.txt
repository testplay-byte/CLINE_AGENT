   145→function resolveRuntimeConfig(
   146→        config: AgentRuntimeConfig,
   147→): BaseAgentRuntimeConfig {
   148→        if (hasPrebuiltModel(config)) {
   149→                return config;
   150→        }
   151→        const { providerId, modelId, apiKey, baseUrl, headers, options, ...rest } =
   152→                config;
   153→        const gateway = createGateway({
   154→                providerConfigs: [{ providerId, apiKey, baseUrl, headers, options }],
   155→                telemetry: rest.telemetry,
   156→        });
   157→        const model = gateway.createAgentModel({ providerId, modelId });
   158→        // The prebuilt-model path preserves a caller-provided messageModelInfo;
   159→        // mirror that here so the provider/model constructor also tags assistant
   160→        // messages with modelInfo. An explicit caller-provided value still wins.
   161→        const messageModelInfo = rest.messageModelInfo ?? {
   162→                id: modelId,
   163→                provider: providerId,
   164→        };
   165→        return { ...rest, model, messageModelInfo };
   166→}

   194→function safeJsonSize(value: unknown): number {
   195→        try {
   196→                return JSON.stringify(value).length;

   209→function summarizeModelRequest(
   210→        request: AgentModelRequest,
   211→): Record<string, unknown> {
   212→        let textChars = request.systemPrompt?.length ?? 0;
   213→        let toolResultCount = 0;
   214→        let toolResultChars = 0;
   215→        let maxToolResultChars = 0;
   216→        for (const message of request.messages) {
   217→                for (const part of message.content) {
   218→                        switch (part.type) {
   219→                                case "text":
   220→                                        textChars += part.text.length;
   221→                                        break;
   222→                                case "reasoning":
   223→                                        textChars += part.text.length;
   224→                                        break;
   225→                                case "file":
   226→                                        textChars += part.content.length;
   227→                                        break;
   228→                                case "tool-call":
   229→                                        textChars += safeJsonSize(part.input);
   230→                                        break;
   231→                                case "tool-result": {
   232→                                        const outputChars = getOutputSize(part.output);
   233→                                        toolResultCount += 1;
   234→                                        toolResultChars += outputChars;
   235→                                        maxToolResultChars = Math.max(maxToolResultChars, outputChars);
   236→                                        textChars += outputChars;
   237→                                        break;
   238→                                }
   239→                        }
   240→                }
   241→        }
   242→
   243→        return {
   244→                messageCount: request.messages.length,
   245→                toolSchemaCount: request.tools.length,
   246→                systemPromptChars: request.systemPrompt?.length ?? 0,
   247→                requestJsonChars: safeJsonSize({
   248→                        systemPrompt: request.systemPrompt,
   249→                        messages: request.messages,
   250→                        tools: request.tools,
   251→                        options: request.options,
   252→                }),
   253→                visibleTextChars: textChars,
   254→                estimatedTextTokens: estimateTokens(textChars),
   255→                toolResultCount,
   256→                toolResultChars,
   257→                maxToolResultChars,
   258→        };
   259→}

   341→function usageDelta(
   342→        start: AgentUsage,
   343→        end: AgentUsage,
   344→): NonNullable<AgentMessage["metrics"]> | undefined {
   345→        const inputTokens = Math.max(
   346→                0,
   347→                (end.inputTokens ?? 0) - (start.inputTokens ?? 0),
   348→        );
   349→        const outputTokens = Math.max(
   350→                0,
   351→                (end.outputTokens ?? 0) - (start.outputTokens ?? 0),
   352→        );
   353→        const cacheReadTokens = Math.max(
   354→                0,
   355→                (end.cacheReadTokens ?? 0) - (start.cacheReadTokens ?? 0),
   356→        );
   357→        const cacheWriteTokens = Math.max(
   358→                0,
   359→                (end.cacheWriteTokens ?? 0) - (start.cacheWriteTokens ?? 0),
   360→        );
   361→        const reasoningTokenCount = Math.max(
   362→                0,
   363→                (end.reasoningTokenCount ?? 0) - (start.reasoningTokenCount ?? 0),
   364→        );
   365→        const startCost = start.totalCost ?? 0;
   366→        const endCost = end.totalCost ?? 0;
   367→        const cost = Math.max(0, endCost - startCost);
   368→        if (
   369→                inputTokens === 0 &&
   370→                outputTokens === 0 &&
   371→                cacheReadTokens === 0 &&
   372→                cacheWriteTokens === 0 &&
   373→                reasoningTokenCount === 0 &&
   374→                cost === 0
   375→        ) {
   376→                return undefined;
   377→        }
   378→        return {
   379→                inputTokens: inputTokens > 0 ? inputTokens : 0,
   380→                outputTokens: outputTokens > 0 ? outputTokens : 0,
   381→                cacheReadTokens: cacheReadTokens > 0 ? cacheReadTokens : 0,
   382→                cacheWriteTokens: cacheWriteTokens > 0 ? cacheWriteTokens : 0,
   383→                ...(reasoningTokenCount > 0 ? { reasoningTokenCount } : {}),
   384→                ...(cost > 0 ? { cost } : {}),
   385→        };
   386→}

   392→function textFromMessage(message: AgentMessage | undefined): string {
   393→        if (!message) {
   394→                return "";
   395→        }
   396→        return message.content
   397→                .filter(
   398→                        (
   399→                                part: AgentMessagePart,
   400→                        ): part is Extract<AgentMessagePart, { type: "text" }> =>
   401→                                part.type === "text",
   402→                )
   403→                .map((part: Extract<AgentMessagePart, { type: "text" }>) => part.text)
   404→                .join("");
   405→}
   406→
   407→function textFromToolMessage(message: AgentMessage | undefined): string {
   408→        const result = message?.content.find(
   409→                (part): part is Extract<AgentMessagePart, { type: "tool-result" }> =>
   410→                        part.type === "tool-result",
   411→        );
   412→        if (!result || result.isError) {
   413→                return "";
   414→        }
   415→        if (typeof result.output === "string") {
   416→                return result.output;
   417→        }
   418→        try {
   419→                return JSON.stringify(result.output);
   420→        } catch {
   421→                return String(result.output);
   422→        }
   423→}

   425→function normalizeInput(input: AgentRunInput): AgentMessage[] {
   426→        if (typeof input === "string") {
   427→                return [createMessage("user", [{ type: "text", text: input }])];
   428→        }
   429→        if (Array.isArray(input)) {
   430→                return cloneMessages(input);
   431→        }
   432→        return cloneMessages([input as AgentMessage]);
   433→}

   435→export class AgentRuntime {
   436→        private config: Required<Pick<BaseAgentRuntimeConfig, "toolExecution">> &
   437→                BaseAgentRuntimeConfig;
   438→        private readonly listeners = new Set<AgentEventListener>();
   439→        // biome-ignore lint/suspicious/noExplicitAny: tool input/output types vary per tool
   440→        private readonly tools = new Map<string, AgentTool<any, any>>();
   441→        private hooks: HookBag = {
   442→                beforeRun: [],
   443→                afterRun: [],
   444→                beforeModel: [],
   445→                afterModel: [],
   446→                beforeTool: [],
   447→                afterTool: [],
   448→                onEvent: [],
   449→        };
   450→        private readonly state = {
   451→                agentId: "",
   452→                agentRole: undefined as string | undefined,
   453→                parentAgentId: undefined as string | null | undefined,
   454→                runId: undefined as string | undefined,
   455→                status: "idle" as AgentRuntimeStateSnapshot["status"],
   456→                iteration: 0,
   457→                messages: [] as AgentMessage[],
   458→                pendingToolCalls: [] as string[],
   459→                usage: cloneUsage(DEFAULT_USAGE),
   460→                lastError: undefined as string | undefined,
   461→                lastErrorClass: undefined as ProviderErrorClass | undefined,
   462→                /**
   463→                 * Whether the model layer already recorded `sdk.error` telemetry for
   464→                 * `lastError` (from `errorReported` on the stream's `finish` event).
   465→                 * Custom `AgentModel` implementations that do not record their own
   466→                 * telemetry leave this false, so their failures still get reported.
   467→                 */
   468→                lastErrorReported: false,
   469→        };
   470→        /** One automatic overflow-recovery attempt per run. */
   471→        private overflowRecoveryAttempted = false;
   472→        private initialization?: Promise<void>;
   473→        private abortController?: AbortController;
   474→        private readonly telemetryProviderId?: string;
   475→        private readonly telemetryModelId?: string;
   476→
   477→        constructor(config: AgentRuntimeConfig) {
   478→                this.telemetryProviderId =
   479→                        trimNonEmpty(config.messageModelInfo?.provider) ??
   480→                        ("providerId" in config ? trimNonEmpty(config.providerId) : undefined);
   481→                this.telemetryModelId =
   482→                        trimNonEmpty(config.messageModelInfo?.id) ??
   483→                        ("modelId" in config ? trimNonEmpty(config.modelId) : undefined);
   484→                const resolved = resolveRuntimeConfig(config);
   485→                this.config = {
   486→                        ...resolved,
   487→                        toolExecution: resolved.toolExecution ?? "sequential",
   488→                };
   489→                this.state.agentId = resolved.agentId ?? createUID("agent");
   490→                this.state.agentRole = resolved.agentRole;
   491→                this.state.parentAgentId = resolved.parentAgentId;
   492→                this.state.messages = cloneMessages(resolved.initialMessages ?? []);
   493→        }
   494→
   495→        async run(input: AgentRunInput): Promise<AgentRunResult> {
   496→                return this.execute(input);
   497→        }
   498→
   499→        async continue(input?: AgentRunInput): Promise<AgentRunResult> {
   500→                return this.execute(input);
   501→        }
   502→
   503→        abort(reason?: unknown): void {
   504→                if (!this.abortController) {
   505→                        return;
   506→                }
   507→                if (this.abortController.signal.aborted) {
   508→                        return;
   509→                }
   510→                const abortError =
   511→                        reason instanceof AgentRuntimeAbortError
   512→                                ? reason
   513→                                : new AgentRuntimeAbortError(reason);
   514→                this.state.lastError = abortError.message;
   515→                this.captureTaskLifecycle(TASK_CANCELLED_EVENT, {
   516→                        error: abortError,
   517→                });
   518→                this.abortController.abort(abortError);
   519→        }
   520→
   521→        subscribe(listener: AgentEventListener): () => void {
   522→                this.listeners.add(listener);
   523→                return () => {
   524→                        this.listeners.delete(listener);
   525→                };
   526→        }
   527→
   528→        /**
   529→         * Replace the conversation with a fresh set of messages, discarding any
   530→         * in-flight run and usage state while preserving the underlying model,
   531→         * tools, hooks, plugins, and active event subscribers.
   532→         *
   533→         * Useful for standalone callers that persist conversations externally and
   534→         * want to re-seed the runtime from storage without recreating subscribers.
   535→         */
   536→        restore(messages: readonly AgentMessage[]): void {
   537→                this.abort("Agent state restored");
   538→                // Reset state that is not carried across restores. Keep `listeners`,
   539→                // tools, hooks, plugins, model, and agent identity so external event
   540→                // subscribers continue to receive events after restore().
   541→                this.state.runId = undefined;
   542→                this.state.status = "idle";
   543→                this.state.iteration = 0;
   544→                this.state.pendingToolCalls = [];
   545→                this.state.usage = cloneUsage(DEFAULT_USAGE);
   546→                this.state.lastError = undefined;
   547→                this.state.lastErrorClass = undefined;
   548→                this.state.lastErrorReported = false;
   549→                this.state.messages = cloneMessages(messages);
   550→                this.config = {
   551→                        ...this.config,
   552→                        initialMessages: cloneMessages(messages),
   553→                };
   554→        }
   555→
   556→        snapshot(): AgentRuntimeStateSnapshot {
   557→                return {
   558→                        agentId: this.state.agentId,
   559→                        agentRole: this.state.agentRole,
   560→                        parentAgentId: this.state.parentAgentId,
   561→                        conversationId: this.config.conversationId?.trim() || undefined,
   562→                        runId: this.state.runId,
   563→                        status: this.state.status,
   564→                        iteration: this.state.iteration,
   565→                        messages: cloneMessages(this.state.messages),
   566→                        pendingToolCalls: [...this.state.pendingToolCalls],
   567→                        usage: cloneUsage(this.state.usage),
   568→                        lastError: this.state.lastError,
   569→                        lastErrorClass: this.state.lastErrorClass,
   570→                };
   571→        }
   572→
   573→        private async ensureInitialized(): Promise<void> {
   574→                this.initialization ??= this.initialize();
   575→                await this.initialization;
   576→        }
   577→
   578→        private async initialize(): Promise<void> {
   579→                this.registerHooks(this.config.hooks);
   580→                for (const tool of this.config.tools ?? []) {
   581→                        this.tools.set(tool.name, tool);
   582→                }
   583→                for (const plugin of this.config.plugins ?? []) {
   584→                        const setup = await plugin.setup?.({
   585→                                agentId: this.state.agentId,
   586→                                agentRole: this.state.agentRole,
   587→                                systemPrompt: this.config.systemPrompt,
   588→                        });
   589→                        for (const tool of setup?.tools ?? []) {
   590→                                this.tools.set(tool.name, tool);
   591→                        }
   592→                        this.registerHooks(setup?.hooks);
   593→                }
   594→        }
   595→
   596→        private registerHooks(hooks: Partial<AgentRuntimeHooks> | undefined): void {
   597→                if (!hooks) {
   598→                        return;
   599→                }
   600→                if (hooks.beforeRun) this.hooks.beforeRun.push(hooks.beforeRun);
   601→                if (hooks.afterRun) this.hooks.afterRun.push(hooks.afterRun);
   602→                if (hooks.beforeModel) this.hooks.beforeModel.push(hooks.beforeModel);
   603→                if (hooks.afterModel) this.hooks.afterModel.push(hooks.afterModel);
   604→                if (hooks.beforeTool) this.hooks.beforeTool.push(hooks.beforeTool);
   605→                if (hooks.afterTool) this.hooks.afterTool.push(hooks.afterTool);
   606→                if (hooks.onEvent) this.hooks.onEvent.push(hooks.onEvent);
   607→        }

   609→        private getRequiredCompletionToolNames(): string[] {
   610→                if (this.config.completionPolicy?.requireCompletionTool !== true) {
   611→                        return [];
   612→                }
   613→                return [...this.tools.values()]
   614→                        .filter((tool) => tool.lifecycle?.completesRun === true)
   615→                        .map((tool) => tool.name)
   616→                        .sort();
   617→        }
   618→
   619→        private getCompletionToolReminderMessage(): string | undefined {
   620→                const terminalToolNames = this.getRequiredCompletionToolNames();
   621→                if (terminalToolNames.length === 0) {
   622→                        return undefined;
   623→                }
   624→                return `[SYSTEM] This run is not complete until you call one of these terminal completion tools: ${terminalToolNames.join(
   625→                        ", ",
   626→                )}. Continue working if requirements are not met. If the task is complete, call the appropriate terminal completion tool now.`;
   627→        }
   628→
   629→        private getCompletionReminderMessages(): string[] {
   630→                return [
   631→                        this.getCompletionToolReminderMessage(),
   632→                        this.config.completionPolicy?.completionGuard?.(),
   633→                ].filter((message): message is string => Boolean(message));
   634→        }
   635→
   636→        private async addUserReminderMessage(text: string): Promise<AgentMessage> {
   637→                const reminderMessage = createMessage("user", [{ type: "text", text }], {
   638→                        userRunSpan: 0,
   639→                });
   640→                this.state.messages.push(reminderMessage);
   641→                await this.emit({
   642→                        type: "message-added",
   643→                        snapshot: this.snapshot(),
   644→                        message: reminderMessage,
   645→                });
   646→                return reminderMessage;
   647→        }
   648→
   649→        private async execute(input?: AgentRunInput): Promise<AgentRunResult> {
   650→                await this.ensureInitialized();
   651→                if (this.state.status === "running") {
   652→                        throw new Error("Agent runtime is already running");
   653→                }
   654→
   655→                this.abortController = new AbortController();
   656→                this.state.runId = createUID("run");
   657→                this.state.status = "running";
   658→                this.state.iteration = 0;
   659→                this.state.pendingToolCalls = [];
   660→                this.state.lastError = undefined;
   661→                this.state.lastErrorClass = undefined;
   662→                this.state.lastErrorReported = false;
   663→                this.state.usage = cloneUsage(DEFAULT_USAGE);
   664→                this.overflowRecoveryAttempted = false;
   665→
   666→                try {
   667→                        await this.callBeforeRunHooks();
   668→                        await this.emit({ type: "run-started", snapshot: this.snapshot() });
   669→
   670→                        for (const message of input ? normalizeInput(input) : []) {
   671→                                this.state.messages.push(message);
   672→                                await this.emit({
   673→                                        type: "message-added",
   674→                                        snapshot: this.snapshot(),
   675→                                        message,
   676→                                });
   677→                        }
   678→
   679→                        const completionToolReminder = this.getCompletionToolReminderMessage();
   680→                        if (completionToolReminder) {
   681→                                await this.addUserReminderMessage(completionToolReminder);
   682→                        }
   683→
   684→                        let finalAssistantMessage: AgentMessage | undefined;
   685→
   686→                        while (
   687→                                this.config.maxIterations === undefined ||
   688→                                this.state.iteration < this.config.maxIterations
   689→                        ) {
   690→                                this.throwIfAborted();
   691→
   692→                                this.state.iteration += 1;
   693→                                await this.emit({
   694→                                        type: "turn-started",
   695→                                        snapshot: this.snapshot(),
   696→                                        iteration: this.state.iteration,
   697→                                });
   698→
   699→                                const { message, finishReason } =
   700→                                        await this.generateAssistantMessageWithOverflowRecovery();
   701→                                if (finishReason === "aborted") {
   702→                                        throw this.normalizeAbortError();
   703→                                }
   704→                                if (message.content.length === 0) {
   705→                                        throw new Error(
   706→                                                finishReason === "error"
   707→                                                        ? (this.state.lastError ?? "Model stream failed")
   708→                                                        : "Model returned empty response",
   709→                                        );
   710→                                }
   711→                                const toolCalls = message.content.filter(
   712→                                        (part: AgentMessagePart): part is AgentToolCallPart =>
   713→                                                part.type === "tool-call",
   714→                                );
   715→
   716→                                finalAssistantMessage = message;
   717→                                this.state.messages.push(message);
   718→                                await this.emit({
   719→                                        type: "message-added",
   720→                                        snapshot: this.snapshot(),
   721→                                        message,
   722→                                });
   723→                                await this.emit({
   724→                                        type: "assistant-message",
   725→                                        snapshot: this.snapshot(),
   726→                                        iteration: this.state.iteration,
   727→                                        message,
   728→                                        finishReason,
   729→                                });
   730→
   731→                                if (finishReason === "max-tokens" && toolCalls.length === 0) {
   732→                                        throw new Error(MAX_TOKENS_INCOMPLETE_TURN_MESSAGE);
   733→                                }
   734→                                if (finishReason === "error" && toolCalls.length === 0) {
   735→                                        throw new Error(this.state.lastError ?? "Model stream failed");
   736→                                }
   737→                                this.state.pendingToolCalls = toolCalls.map((part) => part.toolCallId);
   738→
   739→                                if (toolCalls.length === 0) {
   740→                                        await this.emit({
   741→                                                type: "turn-finished",
   742→                                                snapshot: this.snapshot(),
   743→                                                iteration: this.state.iteration,
   744→                                                toolCallCount: 0,
   745→                                        });
   746→                                        const completionReminderMessages =
   747→                                                this.getCompletionReminderMessages();
   748→                                        if (completionReminderMessages.length > 0) {
   749→                                                for (const reminderMessage of completionReminderMessages) {
   750→                                                        await this.addUserReminderMessage(reminderMessage);
   751→                                                }
   752→                                                continue;
   753→                                        }
   754→                                        const result = this.finishRun("completed", finalAssistantMessage);
   755→                                        await this.callAfterRunHooks(result);
   756→                                        await this.emit({
   757→                                                type: "run-finished",
   758→                                                snapshot: this.snapshot(),
   759→                                                result,
   760→                                        });
   761→                                        return result;
   762→                                }
   763→
   764→                                const toolMessages = await this.executeToolCalls(toolCalls);
   765→                                this.state.pendingToolCalls = [];
   766→                                for (const toolMessage of toolMessages) {
   767→                                        this.state.messages.push(toolMessage);
   768→                                        await this.emit({
   769→                                                type: "message-added",
   770→                                                snapshot: this.snapshot(),
   771→                                                message: toolMessage,
   772→                                        });
   773→                                }
   774→                                await this.emit({
   775→                                        type: "turn-finished",
   776→                                        snapshot: this.snapshot(),
   777→                                        iteration: this.state.iteration,
   778→                                        toolCallCount: toolCalls.length,
   779→                                });
   780→                                const terminalToolMessage = this.findCompletingToolMessage(
   781→                                        toolCalls,
   782→                                        toolMessages,
   783→                                );
   784→                                if (terminalToolMessage) {
   785→                                        const result = this.finishRun(
   786→                                                "completed",
   787→                                                finalAssistantMessage,
   788→                                                textFromToolMessage(terminalToolMessage) || undefined,
   789→                                        );
   790→                                        await this.callAfterRunHooks(result);
   791→                                        await this.emit({
   792→                                                type: "run-finished",
   793→                                                snapshot: this.snapshot(),
   794→                                                result,
   795→                                        });
   796→                                        return result;
   797→                                }
   798→                        }
   799→
   800→                        throw new Error(
   801→                                `Agent runtime exceeded maxIterations (${this.config.maxIterations})`,
   802→                        );
   803→                } catch (error) {
   804→                        const normalized =
   805→                                error instanceof Error ? error : new Error(String(error));
   806→                        const isControlledStop = normalized instanceof ControlledStopError;
   807→                        const isAborted = this.abortController.signal.aborted || isControlledStop;
   808→                        const status = isAborted ? "aborted" : "failed";
   809→                        // Read before overwriting lastError below: the class only applies
   810→                        // when the run failed on the provider error it was recorded for.
   811→                        const errorClass =
   812→                                normalized instanceof ContextWindowOverflowError
   813→                                        ? ("context_window_exceeded" as const)
   814→                                        : normalized.message === this.state.lastError
   815→                                                ? this.state.lastErrorClass
   816→                                                : undefined;
   817→                        // Same guard: the model layer's telemetry only covers this failure
   818→                        // if the run failed on that exact recorded error.
   819→                        const errorAlreadyReported =
   820→                                normalized.message === this.state.lastError &&
   821→                                this.state.lastErrorReported;
   822→                        this.state.status = status;
   823→                        this.state.lastError = normalized.message;
   824→                        this.state.lastErrorClass = errorClass;
   825→                        this.state.lastErrorReported = errorAlreadyReported;
   826→                        const lastAssistantMessage = this.findLastAssistantMessage();
   827→                        const result: AgentRunResult = {
   828→                                agentId: this.state.agentId,
   829→                                agentRole: this.state.agentRole,
   830→                                runId: this.state.runId ?? createUID("run"),
   831→                                status,
   832→                                iterations: this.state.iteration,
   833→                                outputText: textFromMessage(lastAssistantMessage),
   834→                                messages: cloneMessages(this.state.messages),
   835→                                usage: cloneUsage(this.state.usage),
   836→                                error: status === "failed" ? normalized : undefined,
   837→                        };
   838→                        this.config.logger?.log?.("Agent loop caught error", {
   839→                                severity: status === "failed" ? "error" : "warn",
   840→                                agentId: this.state.agentId,
   841→                                agentRole: this.state.agentRole,
   842→                                runId: result.runId,
   843→                                status,
   844→                                iteration: this.state.iteration,
   845→                                errorName: normalized.name,
   846→                                errorMessage: normalized.message,
   847→                                assistantContentPartCount: lastAssistantMessage?.content.length ?? 0,
   848→                        });
   849→                        await this.callAfterRunHooks(result);
   850→                        if (status === "failed") {
   851→                                await this.emit({
   852→                                        type: "run-failed",
   853→                                        snapshot: this.snapshot(),
   854→                                        error: normalized,
   855→                                        errorClass,
   856→                                });
   857→                        } else {
   858→                                await this.emit({
   859→                                        type: "run-finished",
   860→                                        snapshot: this.snapshot(),
   861→                                        result,
   862→                                });
   863→                        }
   864→                        return result;
   865→                } finally {
   866→                        this.abortController = undefined;
   867→                }
   868→        }
   869→
   870→        private async callBeforeRunHooks(): Promise<void> {
   871→                for (const hook of this.hooks.beforeRun) {
   872→                        const control = (await hook({
   873→                                snapshot: this.snapshot(),
   874→                        })) as AgentStopControl | undefined;
   875→                        this.applyStopControl(control);
   876→                }
   877→        }
   878→
   879→        private async callAfterRunHooks(result: AgentRunResult): Promise<void> {
   880→                for (const hook of this.hooks.afterRun) {
   881→                        await hook({ snapshot: this.snapshot(), result });
   882→                }
   883→        }
   884→
   885→        /**
   886→         * Run a model turn, recovering once per run from a provider-rejected
   887→         * context-window overflow: force a compaction through `prepareTurn` and
   888→         * retry the request. Terminal (unrecoverable) overflow states throw with
   889→         * an actionable message instead of the raw provider error.
   890→         */
   891→        private async generateAssistantMessageWithOverflowRecovery(): Promise<{
   892→                message: AgentMessage;
   893→                finishReason: AgentModelFinishReason;
   894→        }> {
   895→                const first = await this.generateAssistantMessage();
   896→                if (!this.isRecoverableOverflowTurn(first)) {
   897→                        return first;
   898→                }
   899→                this.overflowRecoveryAttempted = true;
   900→                const providerError = this.state.lastError;
   901→                if (!this.config.prepareTurn) {
   902→                        throw new ContextWindowOverflowError(
   903→                                CONTEXT_WINDOW_OVERFLOW_NO_RECOVERY_MESSAGE,
   904→                                providerError,
   905→                        );
   906→                }
   907→                await this.emit({
   908→                        type: "status-notice",
   909→                        snapshot: this.snapshot(),
   910→                        message: "context window exceeded — compacting and retrying",
   911→                        metadata: {
   912→                                kind: "context_overflow_recovery",
   913→                                reason: "context_overflow_recovery",
   914→                                phase: "started",
   915→                                iteration: this.state.iteration,
   916→                                providerError,
   917→                        },
   918→                });
   919→                const retry = await this.generateAssistantMessage({
   920→                        overflowRecovery: true,
   921→                });
   922→                if (
   923→                        retry.finishReason === "error" &&
   924→                        this.state.lastErrorClass === "context_window_exceeded"
   925→                ) {
   926→                        throw new ContextWindowOverflowError(
   927→                                CONTEXT_WINDOW_OVERFLOW_RECOVERY_FAILED_MESSAGE,
   928→                                this.state.lastError,
   929→                        );
   930→                }
   931→                return retry;
   932→        }
   933→
   934→        private isRecoverableOverflowTurn(turn: {
   935→                message: AgentMessage;
   936→                finishReason: AgentModelFinishReason;
   937→        }): boolean {
   938→                if (
   939→                        turn.finishReason !== "error" ||
   940→                        this.state.lastErrorClass !== "context_window_exceeded" ||
   941→                        this.overflowRecoveryAttempted
   942→                ) {
   943→                        return false;
   944→                }
   945→                // An errored stream that still produced tool calls proceeds through the
   946→                // normal loop (matching existing behavior); a retry would discard that
   947→                // partial work.
   948→                return !turn.message.content.some((part) => part.type === "tool-call");
   949→        }
   950→
   951→        private async generateAssistantMessage(options?: {
   952→                overflowRecovery?: boolean;
   953→        }): Promise<{
   954→                message: AgentMessage;
   955→                finishReason: AgentModelFinishReason;
   956→        }> {
   957→                const usageBeforeModel = cloneUsage(this.state.usage);
   958→                const modelRequestMetadata = omitUndefinedValues({
   959→                        sessionId: trimNonEmpty(this.config.sessionId),
   960→                        agentId: this.state.agentId,
   961→                        conversationId: trimNonEmpty(this.config.conversationId),
   962→                        runId: this.state.runId,
   963→                        iteration: this.state.iteration,
   964→                });
   965→                let request: AgentModelRequest = {
   966→                        systemPrompt: this.config.systemPrompt,
   967→                        messages: cloneMessages(this.state.messages),
   968→                        tools: [...this.tools.values()].map<AgentToolDefinition>((tool) => ({
   969→                                name: tool.name,
   970→                                description: tool.description,
   971→                                inputSchema: tool.inputSchema,
   972→                        })),
   973→                        signal: this.abortController?.signal,
   974→                        options: mergeModelOptions(this.config.modelOptions, {
   975→                                metadata: modelRequestMetadata,
   976→                        }),
   977→                };
   978→
   979→                const taskLifecycleStartedAt = Date.now();
   980→                const getTaskLifecycleDurationMs = () =>
   981→                        Date.now() - taskLifecycleStartedAt;
   982→
   983→                if (this.state.iteration > 1) {
   984→                        const pendingUserMessage = await this.consumePendingUserMessage();
   985→                        if (pendingUserMessage) {
   986→                                request = {
   987→                                        ...request,
   988→                                        messages: [
   989→                                                ...request.messages,
   990→                                                ...cloneMessages([pendingUserMessage]),
   991→                                        ],
   992→                                };
   993→                        }
   994→                }
   995→
   996→                request = await this.prepareTurnForModelRequest(request, options);
   997→                this.throwIfAborted();
   998→
   999→                for (const hook of this.hooks.beforeModel) {
  1000→                        const result = (await hook({
  1001→                                snapshot: this.snapshot(),
  1002→                                request,
  1003→                        })) as AgentBeforeModelResult | undefined;
  1004→                        this.throwIfAborted();
  1005→                        this.applyStopControl(result);
  1006→                        if (result?.messages) {
  1007→                                request = { ...request, messages: cloneMessages(result.messages) };
  1008→                        }
  1009→                        if (result?.tools) {
  1010→                                request = { ...request, tools: [...result.tools] };
  1011→                        }
  1012→                        if (result?.options) {
  1013→                                request = {
  1014→                                        ...request,
  1015→                                        options: mergeModelOptions(request.options, result.options),
  1016→                                };
  1017→                        }
  1018→                }
  1019→
  1020→                this.config.logger?.debug("Agent model request diagnostics", {
  1021→                        iteration: this.state.iteration,
  1022→                        providerId:
  1023→                                "providerId" in this.config &&
  1024→                                typeof this.config.providerId === "string"
  1025→                                        ? this.config.providerId
  1026→                                        : undefined,
  1027→                        modelId:
  1028→                                "modelId" in this.config && typeof this.config.modelId === "string"
  1029→                                        ? this.config.modelId
  1030→                                        : undefined,
  1031→                        ...summarizeModelRequest(request),
  1032→                });
  1033→
  1034→                this.throwIfAborted();
  1035→                this.captureTaskLifecycle(TASK_PROVIDER_REQUEST_STARTED_EVENT, {
  1036→                        durationMs: getTaskLifecycleDurationMs(),
  1037→                        phase: "provider_request_started",
  1038→                });
  1039→                const stream = this.openTaskLifecycleStream(
  1040→                        request,
  1041→                        getTaskLifecycleDurationMs,
  1042→                );
  1043→
  1044→                const content: AgentMessagePart[] = [];
  1045→                const toolAssemblies = new Map<string, PendingToolAssembly>();
  1046→                const invalidToolCalls: InvalidToolCall[] = [];
  1047→                const sequence: Array<
  1048→                        { type: "tool"; key: string } | { type: "part"; part: AgentMessagePart }
  1049→                > = [];
  1050→                let nextToolIndex = 0;
  1051→                let finishReason: AgentModelFinishReason = "stop";
  1052→                let accumulatedText = "";
  1053→                let accumulatedReasoning = "";
  1054→
  1055→                for await (const event of stream) {
  1056→                        this.throwIfAborted();
  1057→                        switch (event.type) {
  1058→                                case "text-delta": {
  1059→                                        accumulatedText += event.text;
  1060→                                        const last = sequence.at(-1);
  1061→                                        if (last?.type === "part" && last.part.type === "text") {
  1062→                                                last.part.text += event.text;
  1063→                                        } else {
  1064→                                                sequence.push({
  1065→                                                        type: "part",
  1066→                                                        part: { type: "text", text: event.text },
  1067→                                                });
  1068→                                        }
  1069→                                        await this.emit({
  1070→                                                type: "assistant-text-delta",
  1071→                                                snapshot: this.snapshot(),
  1072→                                                iteration: this.state.iteration,
  1073→                                                text: event.text,
  1074→                                                accumulatedText,
  1075→                                        });
  1076→                                        break;
  1077→                                }
  1078→                                case "reasoning-delta": {
  1079→                                        accumulatedReasoning += event.text;
  1080→                                        const last = sequence.at(-1);
  1081→                                        if (last?.type === "part" && last.part.type === "reasoning") {
  1082→                                                last.part.text += event.text;
  1083→                                                last.part.redacted = event.redacted ?? last.part.redacted;
  1084→                                                last.part.metadata = event.metadata ?? last.part.metadata;
  1085→                                        } else {
  1086→                                                sequence.push({
  1087→                                                        type: "part",
  1088→                                                        part: {
  1089→                                                                type: "reasoning",
  1090→                                                                text: event.text,
  1091→                                                                redacted: event.redacted,
  1092→                                                                metadata: event.metadata,
  1093→                                                        },
  1094→                                                });
  1095→                                        }
  1096→                                        await this.emit({
  1097→                                                type: "assistant-reasoning-delta",
  1098→                                                snapshot: this.snapshot(),
  1099→                                                iteration: this.state.iteration,
  1100→                                                text: event.text,
  1101→                                                accumulatedText: accumulatedReasoning,
  1102→                                                redacted: event.redacted,
  1103→                                                metadata: event.metadata,
  1104→                                        });
  1105→                                        break;
  1106→                                }
  1107→                                case "tool-call-delta": {
  1108→                                        const key =
  1109→                                                event.toolCallId ?? `tool_${event.index ?? nextToolIndex}`;
  1110→                                        if (event.index == null && event.toolCallId == null) {
  1111→                                                nextToolIndex += 1;
  1112→                                        }
  1113→                                        let assembly = toolAssemblies.get(key);
  1114→                                        if (!assembly) {
  1115→                                                assembly = {
  1116→                                                        toolCallId: event.toolCallId ?? createUID("tool"),
  1117→                                                        inputText: "",
  1118→                                                };
  1119→                                                toolAssemblies.set(key, assembly);
  1120→                                                sequence.push({ type: "tool", key });
  1121→                                        }
  1122→                                        if (event.toolCallId) {
  1123→                                                assembly.toolCallId = event.toolCallId;
  1124→                                        }
  1125→                                        if (event.toolName) {
  1126→                                                assembly.toolName = event.toolName;
  1127→                                        }
  1128→                                        if (event.input !== undefined) {
  1129→                                                assembly.inputValue = event.input;
  1130→                                        }
  1131→                                        if (event.metadata !== undefined) {
  1132→                                                assembly.metadata = mergeToolMetadata(
  1133→                                                        assembly.metadata,
  1134→                                                        event.metadata,
  1135→                                                );
  1136→                                        }
  1137→                                        if (event.inputText) {
  1138→                                                assembly.inputText = mergeToolInputText(
  1139→                                                        assembly.inputText,
  1140→                                                        event.inputText,
  1141→                                                );
  1142→                                        }
  1143→                                        break;
  1144→                                }
  1145→                                case "file": {
  1146→                                        // Model-generated file output. Preserved into the assistant
  1147→                                        // message so a file-only turn is not treated as empty:
  1148→                                        // images become image parts (the shape providers accept on
  1149→                                        // resend); other media becomes a file part carrying the
  1150→                                        // base64 payload.
  1151→                                        sequence.push({
  1152→                                                type: "part",
  1153→                                                part: event.mediaType.startsWith("image/")
  1154→                                                        ? {
  1155→                                                                        type: "image",
  1156→                                                                        image: event.data,
  1157→                                                                        mediaType: event.mediaType,
  1158→                                                                }
  1159→                                                        : {
  1160→                                                                        type: "file",
  1161→                                                                        path: `model-generated-file-${sequence.length + 1}`,
  1162→                                                                        content: event.data,
  1163→                                                                },
  1164→                                        });
  1165→                                        break;
  1166→                                }
  1167→                                case "usage": {
  1168→                                        await this.updateUsage(event.usage);
  1169→                                        break;
  1170→                                }
  1171→                                case "finish": {
  1172→                                        finishReason = event.reason;
  1173→                                        if (event.error) {
  1174→                                                this.state.lastError = event.error;
  1175→                                                // Models that classify at their own error boundary (where the
  1176→                                                // raw provider error is still structured) win. Anything else —
  1177→                                                // custom `AgentModel` implementations, adapters that carry only
  1178→                                                // a flattened message — is classified from the message so it
  1179→                                                // stays eligible for overflow recovery.
  1180→                                                this.state.lastErrorClass =
  1181→                                                        event.errorClass ?? classifyProviderError(event.error);
  1182→                                                this.state.lastErrorReported = event.errorReported === true;
  1183→                                        }
  1184→                                        break;
  1185→                                }
  1186→                        }
  1187→                }
  1188→
  1189→                for (const item of sequence) {
  1190→                        if (item.type === "part") {
  1191→                                content.push(item.part);
  1192→                                continue;
  1193→                        }
  1194→                        const assembly = toolAssemblies.get(item.key);
  1195→                        if (!assembly?.toolName) {
  1196→                                invalidToolCalls.push({
  1197→                                        toolCallId: assembly?.toolCallId ?? item.key,
  1198→                                        input: buildInvalidToolInput(assembly?.inputText ?? ""),
  1199→                                        reason: "missing_name",
  1200→                                });

  1205→                                invalidToolCalls.push({
  1206→                                        toolCallId: assembly.toolCallId,
  1207→                                        toolName: assembly.toolName,
  1208→                                        input: parsed.invalidInput,
  1209→                                        reason: parsed.reason,
  1210→                                });
  1211→                        }
  1212→                        content.push({
  1213→                                type: "tool-call",
  1214→                                toolCallId: assembly.toolCallId,
  1215→                                toolName: assembly.toolName,
  1216→                                input: parsed.input,
  1217→                                metadata: parsed.parseError
  1218→                                        ? mergeToolMetadata(assembly.metadata, {
  1219→                                                        inputParseError: parsed.parseError,
  1220→                                                        rawInputText: assembly.inputText,
  1221→                                                })
  1222→                                        : assembly.metadata,
  1223→                        });
  1224→                }
  1225→
  1226→                const message = createMessage(
  1227→                        "assistant",
  1228→                        content,
  1229→                        invalidToolCalls.length > 0 ? { invalidToolCalls } : undefined,
  1230→                );
  1231→                const metrics = usageDelta(usageBeforeModel, this.state.usage);
  1232→                if (metrics) {
  1233→                        message.metrics = metrics;
  1234→                        this.captureUnexpectedReasoningTokens(request, metrics);
  1235→                }
  1236→                if (this.config.messageModelInfo) {
  1237→                        message.modelInfo = { ...this.config.messageModelInfo };
  1238→                }
  1239→                for (const hook of this.hooks.afterModel) {
  1240→                        const control = (await hook({
  1241→                                snapshot: this.snapshot(),
  1242→                                assistantMessage: message,
  1243→                                finishReason,
  1244→                        })) as AgentStopControl | undefined;
  1245→                        this.applyStopControl(control);
  1246→                }
  1247→
  1248→                return { message, finishReason };
  1249→        }

  1251→        private async *openTaskLifecycleStream(
  1252→                request: AgentModelRequest,
  1253→                getTaskLifecycleDurationMs: () => number | undefined,
  1254→        ): AsyncIterable<AgentModelEvent> {
  1255→                let stream: AsyncIterable<AgentModelEvent>;

  1257→                try {
  1258→                        stream = await this.config.model.stream(request);

  1271→                                );

  1280→                                        receivedFirstChunk = true;
  1281→                                        phase = "first_chunk_received";

  1283→                                                durationMs: getTaskLifecycleDurationMs(),

  1285→                                                eventType: event.type,
  1286→                                        });
  1287→                                }
  1288→                                yield event;
  1289→                        }
  1290→                } catch (error) {
  1291→                        if (!this.isAbortError(error)) {
  1292→                                this.captureTaskLifecycleFailure(
  1293→                                        error,
  1294→                                        phase,
  1295→                                        getTaskLifecycleDurationMs(),
  1296→                                );
  1297→                        }
  1298→                        throw error;
  1299→                }

  1381→        private async prepareTurnForModelRequest(
  1382→                request: AgentModelRequest,
  1383→                options?: { overflowRecovery?: boolean },
  1384→        ): Promise<AgentModelRequest> {
  1385→                if (!this.config.prepareTurn) {
  1386→                        return request;
  1387→                }
  1388→
  1389→                const overflowRecovery = options?.overflowRecovery === true;
  1390→                const result = await this.config.prepareTurn({
  1391→                        agentId: this.state.agentId,
  1392→                        conversationId: this.config.conversationId,
  1393→                        parentAgentId: this.state.parentAgentId ?? null,
  1394→                        iteration: this.state.iteration,
  1395→                        messages: request.messages,
  1396→                        systemPrompt: request.systemPrompt,
  1397→                        tools: request.tools,
  1398→                        model: {
  1399→                                id: this.config.messageModelInfo?.id,
  1400→                                provider: this.config.messageModelInfo?.provider,
  1401→                        },
  1402→                        signal: request.signal,
  1403→                        overflowRecovery: overflowRecovery || undefined,
  1404→                        emitStatusNotice: (message, metadata) => {
  1405→                                void this.emit({
  1406→                                        type: "status-notice",
  1407→                                        snapshot: this.snapshot(),
  1408→                                        message,
  1409→                                        metadata,
  1410→                                });
  1411→                        },
  1412→                });
  1413→                if (overflowRecovery) {
  1414→                        // Only retry a provider-rejected overflow with a request that is
  1415→                        // actually smaller — anything else is guaranteed to fail again.
  1416→                        //
  1417→                        // Serialized length is a coarse proxy for tokens, which is all this
  1418→                        // backstop needs: it answers "did anything get removed at all" for
  1419→                        // arbitrary `prepareTurn` implementations, and the shared estimator
  1420→                        // is itself linear in character count, so switching units would not
  1421→                        // change the verdict. Authoritative token budgeting (against the
  1422→                        // model's limit) happens inside the compaction pipeline.
  1423→                        // TODO: have `prepareTurn` report the token estimates it already
  1424→                        // computed (before/after) so this decision can use real numbers
  1425→                        // instead of re-deriving a proxy here.
  1426→                        const shrunk =
  1427→                                result?.messages !== undefined &&
  1428→                                JSON.stringify(result.messages).length <
  1429→                                        JSON.stringify(request.messages).length;
  1430→                        if (!shrunk) {
  1431→                                throw new ContextWindowOverflowError(
  1432→                                        CONTEXT_WINDOW_OVERFLOW_NOTHING_TO_COMPACT_MESSAGE,
  1433→                                        this.state.lastError,
  1434→                                );
  1435→                        }
  1436→                }
  1437→                if (!result) {
  1438→                        return request;
  1439→                }
  1440→
  1441→                let next = request;
  1442→                if (result.messages) {
  1443→                        const preparedMessages = cloneMessages(result.messages);
  1444→                        next = { ...next, messages: cloneMessages(preparedMessages) };
  1445→                }
  1446→                if (result.systemPrompt !== undefined) {
  1447→                        next = { ...next, systemPrompt: result.systemPrompt };
  1448→                }
  1449→                return next;
  1450→        }

  1536→        private async prepareToolExecution(
  1537→                toolCall: AgentToolCallPart,
  1538→        ): Promise<PreparedToolExecution> {
  1539→                const tool = this.tools.get(toolCall.toolName);
  1540→                let input = toolCall.input;
  1541→                let skipReason: string | undefined;
  1542→                const metadata =
  1543→                        toolCall.metadata &&
  1544→                        typeof toolCall.metadata === "object" &&
  1545→                        !Array.isArray(toolCall.metadata)
  1546→                                ? (toolCall.metadata as Record<string, unknown>)
  1547→                                : undefined;
  1548→
  1549→                if (typeof metadata?.inputParseError === "string") {
  1550→                        skipReason = metadata.inputParseError;
  1551→                }
  1552→
  1553→                const toolSource =
  1554→                        metadata?.toolSource &&
  1555→                        typeof metadata.toolSource === "object" &&
  1556→                        !Array.isArray(metadata.toolSource)
  1557→                                ? (metadata.toolSource as Record<string, unknown>)
  1558→                                : undefined;
  1559→                if (toolSource?.executionMode === "provider") {
  1560→                        const providerId =
  1561→                                typeof toolSource.providerId === "string"
  1562→                                        ? toolSource.providerId
  1563→                                        : "provider";
  1564→                        skipReason = `Tool execution is disabled for provider ${providerId}`;
  1565→                }
  1566→
  1567→                if (tool && !skipReason) {
  1568→                        input = normalizeJsonLikeStringsForSchema(input, tool.inputSchema);
  1569→                }
  1570→
  1571→                let policyOverride: ToolPolicy | undefined;
  1572→                if (tool && !skipReason) {
  1573→                        for (const hook of this.hooks.beforeTool) {
  1574→                                const result = (await hook({
  1575→                                        snapshot: this.snapshot(),
  1576→                                        tool,
  1577→                                        toolCall: { ...toolCall, input },
  1578→                                        input,
  1579→                                })) as AgentBeforeToolResult | undefined;
  1580→                                if (result?.input !== undefined) {
  1581→                                        input = result.input;
  1582→                                }
  1583→                                if (result?.policy) {
  1584→                                        policyOverride = {
  1585→                                                ...policyOverride,
  1586→                                                ...result.policy,
  1587→                                        };
  1588→                                }
  1589→                                this.applyStopControl(result);
  1590→                                if (result?.skip) {
  1591→                                        skipReason =
  1592→                                                result.reason ?? `Tool ${tool.name} was blocked by a runtime hook`;
  1593→                                        break;
  1594→                                }
  1595→                        }
  1596→                }
  1597→
  1598→                if (tool && !skipReason) {
  1599→                        const policy = {
  1600→                                ...resolveToolPolicy(toolCall.toolName, this.config.toolPolicies),
  1601→                                ...policyOverride,
  1602→                        };
  1603→                        if (policy.enabled === false) {
  1604→                                skipReason = `Tool "${toolCall.toolName}" is disabled by policy`;
  1605→                        } else if (policy.autoApprove === false) {
  1606→                                const approval = await this.requestToolApproval(
  1607→                                        toolCall,
  1608→                                        input,
  1609→                                        policy,
  1610→                                );
  1611→                                if (!approval.approved) {
  1612→                                        skipReason =
  1613→                                                approval.reason ?? `Tool "${toolCall.toolName}" was not approved`;
  1614→                                }
  1615→                        }
  1616→                }
  1617→
  1618→                return {
  1619→                        toolCall: { ...toolCall, input },
  1620→                        tool,
  1621→                        input,
  1622→                        skipReason,
  1623→                };
  1624→        }

  1666→        private async executePreparedTool(
  1667→                prepared: PreparedToolExecution,
  1668→        ): Promise<AgentMessage> {
  1669→                const startedAt = new Date();
  1670→                await this.emit({
  1671→                        type: "tool-started",
  1672→                        snapshot: this.snapshot(),
  1673→                        iteration: this.state.iteration,
  1674→                        toolCall: prepared.toolCall,
  1675→                });
  1676→
  1677→                let result: AgentToolResult;
  1678→                if (prepared.skipReason) {
  1679→                        result = {
  1680→                                output: { error: prepared.skipReason },
  1681→                                isError: true,
  1682→                        };
  1683→                } else if (!prepared.tool) {
  1684→                        result = {
  1685→                                output: { error: `Unknown tool: ${prepared.toolCall.toolName}` },
  1686→                                isError: true,
  1687→                        };
  1688→                } else {
  1689→                        try {
  1690→                                const output = await prepared.tool.execute(prepared.input, {
  1691→                                        sessionId: this.config.sessionId,
  1692→                                        agentId: this.state.agentId,
  1693→                                        conversationId: this.config.conversationId,
  1694→                                        runId: this.state.runId ?? createUID("run"),
  1695→                                        iteration: this.state.iteration,
  1696→                                        toolCallId: prepared.toolCall.toolCallId,
  1697→                                        signal: this.abortController?.signal,
  1698→                                        metadata: this.config.toolContextMetadata,
  1699→                                        snapshot: this.snapshot(),
  1700→                                        emitUpdate: (update: unknown) => {
  1701→                                                void this.emit({
  1702→                                                        type: "tool-updated",
  1703→                                                        snapshot: this.snapshot(),
  1704→                                                        iteration: this.state.iteration,
  1705→                                                        toolCall: prepared.toolCall,
  1706→                                                        update,
  1707→                                                });
  1708→                                        },
  1709→                                });
  1710→                                result = { output };
  1711→                        } catch (error) {
  1712→                                result = {
  1713→                                        output: {
  1714→                                                error: error instanceof Error ? error.message : String(error),
  1715→                                        },
  1716→                                        isError: true,
  1717→                                };
  1718→                        }
  1719→                }
  1720→
  1721→                const endedAt = new Date();
  1722→                const durationMs = Math.max(0, endedAt.getTime() - startedAt.getTime());
  1723→
  1724→                if (prepared.tool) {
  1725→                        for (const hook of this.hooks.afterTool) {
  1726→                                const after = (await hook({
  1727→                                        snapshot: this.snapshot(),
  1728→                                        tool: prepared.tool,
  1729→                                        toolCall: prepared.toolCall,
  1730→                                        input: prepared.input,
  1731→                                        result,
  1732→                                        startedAt,
  1733→                                        endedAt,
  1734→                                        durationMs,
  1735→                                })) as AgentAfterToolResult | undefined;
  1736→                                this.applyStopControl(after);
  1737→                                if (after?.result) {
  1738→                                        result = after.result;
  1739→                                }
  1740→                        }
  1741→                }
  1742→
  1743→                const message = createMessage("tool", [
  1744→                        {
  1745→                                type: "tool-result",
  1746→                                toolCallId: prepared.toolCall.toolCallId,
  1747→                                toolName: prepared.toolCall.toolName,
  1748→                                output: result.output,
  1749→                                isError: result.isError,
  1750→                        },
  1751→                ]);
  1752→
  1753→                await this.emit({
  1754→                        type: "tool-finished",
  1755→                        snapshot: this.snapshot(),
  1756→                        iteration: this.state.iteration,
  1757→                        toolCall: prepared.toolCall,
  1758→                        message,
  1759→                });
  1760→
  1761→                return message;
  1762→        }

  1959→function parseToolArguments(
