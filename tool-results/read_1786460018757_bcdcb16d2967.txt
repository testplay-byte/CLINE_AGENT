    47→ * Resolve the model's usable prompt budget. A reported input limit is
    48→ * authoritative but cannot exceed the context window. When the model only
    49→ * reports a context window, retain a conservative margin for request overhead.
    50→ */
    51→export function resolveEffectiveMaxInputTokens(
    52→        input: Pick<ModelInfo, "maxInputTokens" | "contextWindow">,
    53→): number | undefined {
    54→        const contextWindow = isPositiveFiniteNumber(input.contextWindow)
    55→                ? input.contextWindow
    56→                : undefined;
    57→        const maxInputTokens = isPositiveFiniteNumber(input.maxInputTokens)
    58→                ? input.maxInputTokens
    59→                : undefined;
    60→
    61→        if (maxInputTokens !== undefined) {
    62→                return contextWindow === undefined
    63→                        ? maxInputTokens
    64→                        : Math.min(maxInputTokens, contextWindow);
    65→        }
    66→
    67→        return contextWindow === undefined
    68→                ? undefined
    69→                : contextWindow * CONTEXT_WINDOW_INPUT_RATIO;
    70→}

   134→export function serializeMessage(message: MessageWithMetadata): string {
   135→        if (typeof message.content === "string") {
   136→                return `[${message.role === "user" ? "User" : "Bot"}]: ${message.content}`;
   137→        }
   138→        const lines: string[] = [];
   139→        for (const block of message.content) {
   140→                switch (block.type) {
   141→                        case "text":
   142→                                lines.push(
   143→                                        `[${message.role === "user" ? "User" : "Bot"}]: ${block.text}`,
   144→                                );
   145→                                break;
   146→                        case "thinking":

   148→                                break;

   159→                                break;

   162→                                        `[${message.role === "user" ? "User" : "Bot"} file ${block.path}]: ${truncateText(block.content, FILE_CONTENT_CHAR_LIMIT)}`,
   163→                                );
   164→                                break;
   165→                        case "image":
   166→                                lines.push(
   167→                                        `[${message.role === "user" ? "User" : "Bot"} image]: ${block.mediaType}`,
   168→                                );
   169→                                break;
   170→                }
   171→        }
   172→        return lines.join("\n");
   173→}

   312→/**
   313→ * A cut boundary is safe when starting the preserved tail there cannot
   314→ * orphan half of a tool_use/tool_result pair. Typed user turns qualify,
   315→ * and so do assistant messages: an assistant's tool_use keeps its results
   316→ * in the user message that follows it, so both halves stay on the same
   317→ * side of the cut. A tool_result-only user message is never safe — its
   318→ * matching tool_use sits in the preceding assistant message and would be
   319→ * folded into the summary, leaving an orphaned tool_result the provider
   320→ * rejects.
   321→ */
   322→function isSafeCutBoundary(message: MessageWithMetadata): boolean {
   323→        return message.role === "assistant" || isTurnStartMessage(message);
   324→}

   326→export function findCutIndex(
   327→        messages: MessageWithMetadata[],
   328→        preserveRecentTokens: number,
   329→        estimateMessageTokens: EstimateMessageTokens,
   330→): number {
   331→        let total = 0;
   332→        let candidate = messages.length;
   333→        for (let index = messages.length - 1; index >= 0; index -= 1) {
   334→                total += estimateMessageTokens(messages[index]);
   335→                candidate = index;
   336→                if (total >= preserveRecentTokens) {
   337→                        break;
   338→                }
   339→        }
   340→
   341→        if (candidate <= 0) {
   342→                return 0;
   343→        }
   344→
   345→        // Never summarize away the latest typed user prompt: when one exists
   346→        // past index 0, the cut stays at or before it so that whole turn
   347→        // survives verbatim. Transcripts without a later typed turn (a single
   348→        // task followed by one long tool loop, or a projection that starts
   349→        // with a compaction summary) still cut at the token-budget candidate.
   350→        const lastTurnStartIndex = findLastTurnStartIndex(messages);
   351→        let cut =
   352→                lastTurnStartIndex > 0
   353→                        ? Math.min(candidate, lastTurnStartIndex)
   354→                        : candidate;
   355→        while (cut > 0 && !isSafeCutBoundary(messages[cut])) {
   356→                cut -= 1;
   357→        }
   358→        return cut;
   359→}

   361→export function collectPaths(value: unknown): string[] {
   362→        if (typeof value === "string" && value.trim().length > 0) {
   363→                return [value];
   364→        }
   365→        if (Array.isArray(value)) {
   366→                return value.flatMap((item) => collectPaths(item));
   367→        }
   368→        if (value && typeof value === "object") {
   369→                const record = value as Record<string, unknown>;
   370→                const paths: string[] = [];
   371→                for (const key of [
   372→                        "path",
   373→                        "file_path",
   374→                        "target_file",
   375→                        "new_file_path",
   376→                        "old_file_path",
   377→                ]) {
   378→                        paths.push(...collectPaths(record[key]));
   379→                }
   380→                if (Array.isArray(record.files)) {
   381→                        for (const item of record.files) {
   382→                                if (item && typeof item === "object") {
   383→                                        paths.push(...collectPaths((item as Record<string, unknown>).path));
   384→                                }
   385→                        }
   386→                }
   387→                if (Array.isArray(record.file_paths)) {
   388→                        paths.push(...collectPaths(record.file_paths));
   389→                }
   390→                return paths;
   391→        }
   392→        return [];
   393→}

   395→export function mergeUnique(base: string[], next: Iterable<string>): string[] {
   396→        const seen = new Set(base);
   397→        for (const value of next) {
   398→                const trimmed = value.trim();
   399→                if (!trimmed) {
   400→                        continue;
   401→                }
   402→                seen.add(trimmed);
   403→        }
   404→        return [...seen].sort((a, b) => a.localeCompare(b));
   405→}

   407→export function extractFileOps(
   408→        messages: MessageWithMetadata[],
   409→): FileOperationSummary {
   410→        let readFiles: string[] = [];
   411→        let modifiedFiles: string[] = [];
   412→        for (const message of messages) {
   413→                const summaryMetadata = getCompactionSummaryMetadata(message);
   414→                if (summaryMetadata) {
   415→                        readFiles = mergeUnique(readFiles, summaryMetadata.details.readFiles);
   416→                        modifiedFiles = mergeUnique(
   417→                                modifiedFiles,
   418→                                summaryMetadata.details.modifiedFiles,
   419→                        );
   420→                        continue;
   421→                }
   422→                if (!Array.isArray(message.content)) {
   423→                        continue;
   424→                }
   425→                for (const block of message.content) {
   426→                        if (block.type === "file") {
   427→                                readFiles = mergeUnique(readFiles, [block.path]);
   428→                                continue;

   437→                        }
   438→                        if (block.name === "editor" || block.name === "apply_patch") {
   439→                                modifiedFiles = mergeUnique(modifiedFiles, paths);

   441→                }
   442→        }

   482→function formatReadFileEntry(
   483→        record: Record<string, unknown>,
   484→): string | undefined {
   485→        const path = typeof record.path === "string" ? record.path.trim() : "";
   486→        if (!path) {
   487→                return undefined;
   488→        }
   489→        const start = record.start_line;
   490→        const end = record.end_line;
   491→        if (typeof start === "number" && typeof end === "number") {
   492→                return `${path}:${start}-${end}`;
   493→        }
   494→        return path;
   495→}

   497→/**
   498→ * Pull the edited line range out of an editor tool result. Editor results
   499→ * embed a numbered diff (`-467: ...` / `+467: ...`), usually inside a
   500→ * JSON-encoded payload's `result` field; the range is the min/max line
   501→ * number seen. Returns undefined when no diff line numbers are found.
   502→ */
   503→function extractDiffLineRange(
   504→        content: ToolResultContent["content"],
   505→): { start: number; end: number } | undefined {
   506→        let text =
   507→                typeof content === "string"
   508→                        ? content
   509→                        : content
   510→                                        .map((block) =>
   511→                                                block.type === "text"
   512→                                                        ? block.text
   513→                                                        : block.type === "file"
   514→                                                                ? block.content
   515→                                                                : "",
   516→                                        )
   517→                                        .join("\n");
   518→        try {
   519→                const parsed = JSON.parse(text) as { result?: unknown };
   520→                if (parsed && typeof parsed.result === "string") {
   521→                        text = parsed.result;
   522→                }
   523→        } catch {
   524→                // Not JSON-encoded; scan the raw text.
   525→        }
   526→        let start = Number.POSITIVE_INFINITY;
   527→        let end = Number.NEGATIVE_INFINITY;
   528→        for (const match of text.matchAll(/(?:^|\n|\\n)[-+](\d+): /g)) {
   529→                const line = Number(match[1]);
   530→                if (!Number.isFinite(line)) {
   531→                        continue;
   532→                }
   533→                start = Math.min(start, line);
   534→                end = Math.max(end, line);
   535→        }
   536→        return start <= end ? { start, end } : undefined;
   537→}

   539→/**
   540→ * Summarize the tool work inside a span of messages that compaction is
   541→ * dropping. Read ranges come from read_files inputs; edited ranges come
   542→ * from the numbered diff in the matching editor result when present.
   543→ */
   544→export function summarizeToolActivity(
   545→        messages: MessageWithMetadata[],
   546→): ToolActivitySummary {
   547→        const readFiles: string[] = [];
   548→        const editedFiles: string[] = [];
   549→        const commands: string[] = [];
   550→        const editorPathsByToolUseId = new Map<string, string>();
   551→        for (const message of messages) {
   552→                if (!Array.isArray(message.content)) {
   553→                        continue;
   554→                }
   555→                for (const block of message.content) {
   556→                        if (block.type === "tool_use") {
   557→                                const input = (block.input ?? {}) as Record<string, unknown>;
   558→                                if (block.name === "read_files") {
   559→                                        if (Array.isArray(input.files)) {
   560→                                                for (const file of input.files) {
   561→                                                        if (file && typeof file === "object") {
   562→                                                                const entry = formatReadFileEntry(
   563→                                                                        file as Record<string, unknown>,
   564→                                                                );
   565→                                                                if (entry) {
   566→                                                                        pushUniqueEntry(readFiles, entry);
   567→                                                                }
   568→                                                        }
   569→                                                }
   570→                                        } else {
   571→                                                for (const path of collectPaths(input)) {
   572→                                                        pushUniqueEntry(readFiles, path);
   573→                                                }
   574→                                        }
   575→                                        continue;
   576→                                }
   577→                                if (block.name === "editor" || block.name === "apply_patch") {
   578→                                        const path = collectPaths(input)[0];
   579→                                        if (path) {
   580→                                                editorPathsByToolUseId.set(block.id, path);
   581→                                        }
   582→                                        continue;
   583→                                }
   584→                                if (block.name === "run_commands") {
   585→                                        const commandList = Array.isArray(input.commands)
   586→                                                ? input.commands
   587→                                                : [];
   588→                                        for (const command of commandList) {
   589→                                                if (typeof command === "string" && command.trim()) {
   590→                                                        pushUniqueEntry(
   591→                                                                commands,
   592→                                                                truncateCommandForSummary(command.trim()),
   593→                                                        );
   594→                                                }
   595→                                        }
   596→                                }
   597→                                continue;
   598→                        }
   599→                        if (block.type === "tool_result") {
   600→                                const path = editorPathsByToolUseId.get(block.tool_use_id);
   601→                                if (!path) {
   602→                                        continue;
   603→                                }
   604→                                const range = extractDiffLineRange(block.content);
   605→                                pushUniqueEntry(
   606→                                        editedFiles,
   607→                                        range ? `${path}:${range.start}-${range.end}` : path,
   608→                                );
   609→                                editorPathsByToolUseId.delete(block.tool_use_id);
   610→                        }
   611→                }
   612→        }
   613→        // Editor calls whose results fell outside the span still count as edits.
   614→        for (const path of editorPathsByToolUseId.values()) {
   615→                pushUniqueEntry(editedFiles, path);
   616→        }
   617→        return { readFiles, editedFiles, commands };
   618→}

   684→
   685→        return parts.join("\n\n");
   686→}
   687→
   688→/**
   689→ * The summarizer output budget is a cap, not a target: reasoning models need
   690→ * headroom beyond their thinking output or no summary text ever arrives and
   691→ * compaction is skipped. An explicit configuration wins as-is; otherwise the
   692→ * default applies, with model metadata (`maxTokens` is reported capability,
   693→ * not a product default) only ever lowering it.
   694→ */
   695→function resolveSummaryMaxOutputTokens(config: ProviderConfig): number {
   696→        if (isPositiveFiniteNumber(config.maxOutputTokens)) {
   697→                return Math.floor(config.maxOutputTokens);
   698→        }
   699→        const modelMaxTokens =
   700→                config.modelInfo?.maxTokens ??
   701→                config.knownModels?.[config.modelId]?.maxTokens;
   702→        if (isPositiveFiniteNumber(modelMaxTokens)) {
   703→                return Math.min(
   704→                        DEFAULT_SUMMARY_MAX_OUTPUT_TOKENS,
   705→                        Math.floor(modelMaxTokens),
   706→                );
   707→        }
   708→        return DEFAULT_SUMMARY_MAX_OUTPUT_TOKENS;
   709→}
   710→
   711→export function resolveSummarizerConfig(options: {
   712→        activeProviderConfig: ProviderConfig;
   713→        summarizer?: CoreCompactionSummarizerConfig;
   714→}): ProviderConfig {
   715→        const summarizer = options.summarizer;
   716→        const withSummarizerDefaults = (config: ProviderConfig): ProviderConfig => {
   717→                if (config.providerId === "openai-codex") {
   718→                        const { maxOutputTokens: _maxOutputTokens, ...rest } = config;
   719→                        return {
   720→                                ...rest,
   721→                                thinking: false,
   722→                        };
   723→                }
   724→                return {
   725→                        ...config,
   726→                        maxOutputTokens: resolveSummaryMaxOutputTokens(config),
   727→                        thinking: false,
   728→                };
   729→        };
   730→        if (!summarizer) {
   731→                return withSummarizerDefaults(options.activeProviderConfig);
   732→        }
   733→        const baseProviderConfig =
   734→                summarizer.providerConfig?.providerId === summarizer.providerId
   735→                        ? summarizer.providerConfig
   736→                        : undefined;
   737→        return withSummarizerDefaults({
   738→                ...(baseProviderConfig ?? {}),
   739→                providerId: summarizer.providerId,
   740→                modelId: summarizer.modelId,
   741→                apiKey: summarizer.apiKey ?? baseProviderConfig?.apiKey,
   742→                baseUrl: summarizer.baseUrl ?? baseProviderConfig?.baseUrl,
   743→                headers: summarizer.headers ?? baseProviderConfig?.headers,
   744→                modelInfo: summarizer.modelInfo ?? baseProviderConfig?.modelInfo,
   745→                knownModels: summarizer.knownModels ?? baseProviderConfig?.knownModels,
   746→                maxOutputTokens:
   747→                        summarizer.maxOutputTokens ?? baseProviderConfig?.maxOutputTokens,
   748→        });
   749→}
