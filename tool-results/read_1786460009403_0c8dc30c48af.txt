    91→function isCompactionCancellation(
    92→        error: unknown,
    93→        abortSignal: AbortSignal,
    94→): boolean {
    95→        if (abortSignal.aborted) {
    96→                return true;
    97→        }
    98→        return (
    99→                error instanceof Error &&
   100→                (error.name === "AbortError" || error.name === "AgentRuntimeAbortError")
   101→        );
   102→}
   103→
   104→function describeCompactionError(error: unknown): Record<string, unknown> {
   105→        return error instanceof Error
   106→                ? { errorName: error.name, errorMessage: error.message }
   107→                : { errorMessage: String(error) };
   108→}
   109→
   110→function safeJsonSize(value: unknown): number {
   111→        try {
   112→                return JSON.stringify(value).length;
   113→        } catch {
   114→                return String(value).length;
   115→        }
   116→}
   117→
   118→function summarizeToolResults(messages: CoreCompactionContext["messages"]): {
   119→        toolResultCount: number;
   120→        toolResultSerializedChars: number;
   121→        maxToolResultSerializedChars: number;
   122→} {
   123→        let toolResultCount = 0;
   124→        let toolResultSerializedChars = 0;
   125→        let maxToolResultSerializedChars = 0;
   126→        for (const message of messages) {
   127→                if (!Array.isArray(message.content)) {
   128→                        continue;
   129→                }
   130→                for (const block of message.content) {
   131→                        if (block.type !== "tool_result") {
   132→                                continue;
   133→                        }
   134→                        const size = safeJsonSize(block.content);
   135→                        toolResultCount += 1;
   136→                        toolResultSerializedChars += size;
   137→                        maxToolResultSerializedChars = Math.max(
   138→                                maxToolResultSerializedChars,
   139→                                size,
   140→                        );
   141→                }
   142→        }
   143→        return {
   144→                toolResultCount,
   145→                toolResultSerializedChars,
   146→                maxToolResultSerializedChars,
   147→        };
   148→}

   151→        basic: ({ context, estimateMessageTokens, logger }) =>

   177→function resolveManualMessageTargetTokens(input: {
   178→        messageInputTokens: number;
   179→        messageTriggerTokens: number;
   180→        manualTargetRatio: number | undefined;
   181→}): number {
   182→        const ratio =
   183→                typeof input.manualTargetRatio === "number" &&
   184→                Number.isFinite(input.manualTargetRatio)
   185→                        ? input.manualTargetRatio
   186→                        : 0.5;
   187→        const targetRatio = Math.min(0.95, Math.max(0.05, ratio));
   188→        return Math.max(
   189→                1,
   190→                Math.floor(
   191→                        Math.min(
   192→                                input.messageTriggerTokens,
   193→                                input.messageInputTokens * targetRatio,
   194→                        ),
   195→                ),
   196→        );
   197→}
   198→
   199→function resolveAutoRequestTargetTokens(input: {
   200→        maxInputTokens: number;
   201→        modelMaxTokens?: number;
   202→        triggerTokens: number;
   203→        messagePairCount: number;
   204→}): number {
   205→        const targetTokens =
   206→                input.messagePairCount >= 5 &&
   207→                typeof input.modelMaxTokens === "number" &&
   208→                Number.isFinite(input.modelMaxTokens) &&
   209→                input.modelMaxTokens < input.maxInputTokens
   210→                        ? Math.floor(input.maxInputTokens * LONG_CONVERSATION_TARGET_RATIO)
   211→                        : Math.floor(input.triggerTokens * DEFAULT_TARGET_RATIO);
   212→        const triggerCeiling = Math.max(1, input.triggerTokens - 1);
   213→        return Math.max(
   214→                1,
   215→                Math.min(targetTokens, input.maxInputTokens, triggerCeiling),
   216→        );
   217→}
   218→
   219→function translateRequestBudgetToMessages(
   220→        requestTokens: number,
   221→        overheadTokens: number,
   222→): number {
   223→        return Math.max(1, Math.floor(requestTokens - overheadTokens));
   224→}
   225→
   226→function countUserAssistantPairs(
   227→        messages: CoreCompactionContext["messages"],
   228→): number {
   229→        let pairs = 0;
   230→        let hasPendingUser = false;
   231→        for (const message of messages) {
   232→                if (message.role === "user") {
   233→                        hasPendingUser = true;
   234→                } else if (message.role === "assistant" && hasPendingUser) {
   235→                        pairs += 1;
   236→                        hasPendingUser = false;
   237→                }
   238→        }
   239→        return pairs;
   240→}

   242→/**
   243→ * Build the `prepareTurn` callback used by the agent runtime to compact the
   244→ * transcript before each model request.
   245→ *
   246→ * Telemetry: emits `task.compaction_executed` on a successful compaction and
   247→ * `task.compaction_skipped` when the configured strategy returns `undefined`.
   248→ * Telemetry is keyed by `config.sessionId` (falling back to the per-turn
   249→ * `conversationId`) and tagged with `provider` / `modelId`.
   250→ *
   251→ * Known gap: compactions performed via plugin `registerMessageBuilder()` or
   252→ * via the `beforeModel` runtime hook bypass this wrapper entirely, so they
   253→ * do not emit compaction telemetry. If we want coverage there too, the
   254→ * plugin/hook pipelines must be instrumented separately.
   255→ */
   256→export function createContextCompactionPrepareTurn(
   257→        config: Pick<
   258→                CoreSessionConfig,
   259→                | "providerConfig"
   260→                | "providerId"
   261→                | "modelId"
   262→                | "compaction"
   263→                | "logger"
   264→                | "telemetry"
   265→                | "sessionId"
   266→        >,
   267→        options: ContextCompactionPrepareTurnOptions = {},
   268→):
   269→        | ((
   270→                        context: ContextPipelinePrepareTurnInput,
   271→          ) => Promise<ContextPipelinePrepareTurnResult | undefined>)
   272→        | undefined {
   273→        const userCompaction = config.compaction;
   274→        if (userCompaction?.enabled !== true) {
   275→                return undefined;
   276→        }
   277→
   278→        const providerConfig =
   279→                config.providerConfig ??
   280→                ({
   281→                        providerId: config.providerId,
   282→                        modelId: config.modelId,
   283→                } as ProviderConfig);
   284→        const estimateMessageTokens = createTokenEstimator();
   285→        const strategy = userCompaction?.strategy ?? "agentic";
   286→        const runBuiltinStrategy = BUILTIN_COMPACTION_STRATEGIES[strategy];
   287→        const mode = options.mode ?? "auto";
   288→        const telemetryStrategy: TelemetryCompactionStrategy = userCompaction?.compact
   289→                ? "custom"
   290→                : strategy;
   291→
   292→        return async (context) => {
   293→                const effectiveMode: CoreCompactionMode = context.overflowRecovery
   294→                        ? "overflow_recovery"
   295→                        : mode;
   296→                const apiMessageTokens = context.apiMessages.reduce(
   297→                        (total: number, message) => total + estimateMessageTokens(message),
   298→                        0,
   299→                );
   300→                const requestInputTokens = estimateRequestInputTokens({
   301→                        systemPrompt: context.systemPrompt,
   302→                        messages: context.apiMessages,
   303→                        tools: context.tools,
   304→                });
   305→                const messageInputTokens = context.messages.reduce(
   306→                        (total: number, message) => total + estimateMessageTokens(message),
   307→                        0,
   308→                );
   309→                const requestOverheadTokens = Math.max(
   310→                        0,
   311→                        requestInputTokens - apiMessageTokens,
   312→                );
   313→                const maxInputTokens =
   314→                        resolveEffectiveMaxInputTokens({
   315→                                maxInputTokens: context.model.info?.maxInputTokens,
   316→                                contextWindow: context.model.info?.contextWindow,
   317→                        }) ?? DEFAULT_MAX_INPUT_TOKENS;
   318→                const requestTriggerTokens = maxInputTokens * COMPACTION_TRIGGER_RATIO;
   319→                const messageTriggerTokens = translateRequestBudgetToMessages(
   320→                        requestTriggerTokens,
   321→                        requestOverheadTokens,
   322→                );
   323→                const shouldCompact = requestInputTokens >= requestTriggerTokens;
   324→                config.logger?.debug("Context compaction diagnostics", {
   325→                        mode: effectiveMode,
   326→                        strategy,
   327→                        iteration: context.iteration,
   328→                        providerId: config.providerId,
   329→                        modelId: config.modelId,
   330→                        requestInputTokens,
   331→                        apiMessageTokens,
   332→                        messageInputTokens,
   333→                        requestOverheadTokens,
   334→                        maxInputTokens,
   335→                        requestTriggerTokens,
   336→                        messageTriggerTokens,
   337→                        thresholdRatio: COMPACTION_TRIGGER_RATIO,
   338→                        shouldCompact,
   339→                        messageCount: context.messages.length,
   340→                        apiMessageCount: context.apiMessages.length,
   341→                        apiMessagesJsonChars: safeJsonSize(context.apiMessages),
   342→                        ...summarizeToolResults(context.apiMessages),
   343→                });
   344→                if (effectiveMode === "auto" && !shouldCompact) {
   345→                        return undefined;
   346→                }
   347→                let requestTargetTokens: number;
   348→                let messageTargetTokens: number;
   349→                if (effectiveMode === "auto") {
   350→                        requestTargetTokens = resolveAutoRequestTargetTokens({
   351→                                maxInputTokens,
   352→                                modelMaxTokens: context.model.info?.maxTokens,
   353→                                triggerTokens: requestTriggerTokens,
   354→                                messagePairCount: countUserAssistantPairs(context.messages),
   355→                        });
   356→                        messageTargetTokens = translateRequestBudgetToMessages(
   357→                                requestTargetTokens,
   358→                                requestOverheadTokens,
   359→                        );
   360→                } else {
   361→                        messageTargetTokens = resolveManualMessageTargetTokens({
   362→                                messageInputTokens,
   363→                                messageTriggerTokens,
   364→                                manualTargetRatio: options.manualTargetRatio,
   365→                        });
   366→                        requestTargetTokens = requestOverheadTokens + messageTargetTokens;
   367→                }
   368→
   369→                const compactionContext = {
   370→                        agentId: context.agentId,
   371→                        conversationId: context.conversationId,
   372→                        parentAgentId: context.parentAgentId,
   373→                        iteration: context.iteration,
   374→                        messages: context.messages,
   375→                        model: context.model,
   376→                        mode: effectiveMode,
   377→                        abortSignal: context.abortSignal,
   378→                        budget: {
   379→                                request: {
   380→                                        inputTokens: requestInputTokens,
   381→                                        maxInputTokens,
   382→                                        triggerTokens: requestTriggerTokens,
   383→                                        targetTokens: requestTargetTokens,
   384→                                        overheadTokens: requestOverheadTokens,
   385→                                        thresholdRatio: COMPACTION_TRIGGER_RATIO,
   386→                                        utilizationRatio:
   387→                                                maxInputTokens > 0 ? requestInputTokens / maxInputTokens : 0,
   388→                                },
   389→                                messages: {
   390→                                        inputTokens: messageInputTokens,
   391→                                        triggerTokens: messageTriggerTokens,
   392→                                        targetTokens: messageTargetTokens,
   393→                                },
   394→                        },
   395→                };
   396→
   397→                const statusReason =
   398→                        effectiveMode === "manual"
   399→                                ? "manual_compaction"
   400→                                : effectiveMode === "overflow_recovery"
   401→                                        ? "overflow_recovery_compaction"
   402→                                        : "auto_compaction";
   403→                const noticePrefix =
   404→                        effectiveMode === "manual"
   405→                                ? ""
   406→                                : effectiveMode === "overflow_recovery"
   407→                                        ? "overflow-recovery-"
   408→                                        : "auto-";
   409→                context.emitStatusNotice?.(`${noticePrefix}compacting`, {
   410→                        kind: statusReason,
   411→                        reason: statusReason,
   412→                        phase: "started",
   413→                        iteration: context.iteration,
   414→                        triggerTokens: requestTriggerTokens,
   415→                        targetTokens: requestTargetTokens,
   416→                        maxInputTokens,
   417→                        messageTargetTokens,
   418→                });
   419→
   420→                const beforeMessageCount = context.messages.length;
   421→                const startedAt = Date.now();
   422→
   423→                const builtinOptions = {
   424→                        context: compactionContext,
   425→                        providerConfig: {
   426→                                ...providerConfig,
   427→                                abortSignal: context.abortSignal,
   428→                        },
   429→                        compaction: userCompaction,
   430→                        estimateMessageTokens,
   431→                        logger: config.logger,
   432→                };
   433→                let executedStrategy = telemetryStrategy;
   434→                let result: CoreCompactionResult | undefined;
   435→                if (effectiveMode === "overflow_recovery") {
   436→                        // The provider already rejected the request, so recovery must end
   437→                        // deterministically: the agentic strategy's own summarizer call could
   438→                        // overflow the same window (its input budgeting trusts the same
   439→                        // estimator that just undercounted). A custom compactor gets first
   440→                        // shot — it sees mode "overflow_recovery" and owns its transcript
   441→                        // invariants — but its result is held to the same bar basic
   442→                        // compaction aims for: strictly smaller than the input (the runtime
   443→                        // refuses to retry with a request that is not smaller) AND within
   444→                        // the recovery token target. A marginal shrink would spend the
   445→                        // run's single retry on a request that still cannot fit. On throw,
   446→                        // decline, or an insufficient result, basic compaction runs so
   447→                        // recovery never depends on another successful LLM request.
   448→                        if (userCompaction?.compact) {
   449→                                try {
   450→                                        result = await userCompaction.compact(compactionContext);
   451→                                } catch (error) {
   452→                                        if (isCompactionCancellation(error, context.abortSignal)) {
   453→                                                throw error;
   454→                                        }
   455→                                        config.logger?.log(
   456→                                                "Custom compaction failed during overflow recovery; falling back to basic compaction",
   457→                                                {
   458→                                                        severity: "warn",
   459→                                                        ...describeCompactionError(error),
   460→                                                },
   461→                                        );
   462→                                        result = undefined;
   463→                                }
   464→                                if (result?.messages) {
   465→                                        const customMessageTokens = result.messages.reduce(
   466→                                                (total: number, message) => total + estimateMessageTokens(message),
   467→                                                0,
   468→                                        );
   469→                                        // The full acceptance bar, covering every degenerate size: a
   470→                                        // non-empty transcript (an empty one erases the request being
   471→                                        // retried), strictly smaller than the input (the runtime
   472→                                        // refuses a retry that is not smaller), and within the
   473→                                        // recovery token target (a marginal shrink spends the run's
   474→                                        // single retry on a request that still cannot fit). Both size
   475→                                        // comparisons use the token estimator rather than serialized
   476→                                        // length so they are expressed in the same unit as the target.
   477→                                        const acceptable =
   478→                                                result.messages.length > 0 &&
   479→                                                customMessageTokens < messageInputTokens &&
   480→                                                customMessageTokens <= messageTargetTokens;
   481→                                        if (!acceptable) {
   482→                                                config.logger?.log(
   483→                                                        "Custom compaction did not produce an acceptable overflow-recovery transcript; falling back to basic compaction",
   484→                                                        {
   485→                                                                severity: "warn",
   486→                                                                customMessageCount: result.messages.length,
   487→                                                                customMessageTokens,
   488→                                                                messageTargetTokens,
   489→                                                        },
   490→                                                );
   491→                                                result = undefined;
   492→                                        }
   493→                                }
   494→                        }
   495→                        if (!result?.messages) {
   496→                                executedStrategy = "basic";
   497→                                result = await BUILTIN_COMPACTION_STRATEGIES.basic(builtinOptions);
   498→                        }
   499→                } else if (userCompaction?.compact) {
   500→                        result = await userCompaction.compact(compactionContext);
   501→                } else {
   502→                        try {
   503→                                result = await runBuiltinStrategy(builtinOptions);
   504→                        } catch (error) {
   505→                                if (
   506→                                        strategy !== "agentic" ||
   507→                                        isCompactionCancellation(error, context.abortSignal)
   508→                                ) {
   509→                                        throw error;
   510→                                }
   511→                                config.logger?.log(
   512→                                        "Agentic compaction failed; falling back to basic compaction",
   513→                                        {
   514→                                                severity: "warn",
   515→                                                ...describeCompactionError(error),
   516→                                        },
   517→                                );
   518→                                executedStrategy = "basic";
   519→                                result = await BUILTIN_COMPACTION_STRATEGIES.basic(builtinOptions);
   520→                        }
   521→                }
   522→
   523→                const durationMs = Date.now() - startedAt;
   524→                // Telemetry identity: surface the agent/conversation passed into the
   525→                // prepareTurn so multi-agent runs can attribute compactions correctly.
   526→                // `sessionId` is the host-owned session id (ulid). We fall back to the
   527→                // conversation id when no sessionId is supplied (e.g. ad-hoc callers).
   528→                const telemetryUlid = config.sessionId ?? context.conversationId;
   529→                const telemetryIdentity = {
   530→                        agentId: context.agentId,
   531→                        conversationId: context.conversationId,
   532→                        parentAgentId: context.parentAgentId ?? undefined,
   533→                };
   534→
   535→                if (result?.messages) {
   536→                        const afterMessageTokens = result.messages.reduce(
   537→                                (total: number, message) => total + estimateMessageTokens(message),
   538→                                0,
   539→                        );
   540→                        const afterRequestTokens = requestOverheadTokens + afterMessageTokens;
   541→                        config.logger?.log("Context compaction completed", {
   542→                                severity: "info",
   543→                                strategy: executedStrategy,
   544→                                maxInputTokens,
   545→                                messageInputTokens,
   546→                                apiInputTokens: apiMessageTokens,
   547→                                requestInputTokens,
   548→                                requestOverheadTokens,
   549→                                afterMessageTokens,
   550→                                afterRequestTokens,
   551→                                tokensSaved: requestInputTokens - afterRequestTokens,
   552→                                utilizationBefore: `${((requestInputTokens / maxInputTokens) * 100).toFixed(1)}%`,
   553→                                utilizationAfter: `${((afterRequestTokens / maxInputTokens) * 100).toFixed(1)}%`,
   554→                                thresholdTrigger: `${(COMPACTION_TRIGGER_RATIO * 100).toFixed(1)}%`,
   555→                                messagesBefore: beforeMessageCount,
   556→                                messagesAfter: result.messages.length,
   557→                                messagesRemoved: beforeMessageCount - result.messages.length,
   558→                        } as Record<string, unknown>);
   559→                        context.emitStatusNotice?.(`${noticePrefix}compacted`, {
   560→                                kind: statusReason,
   561→                                reason: statusReason,
   562→                                phase: "completed",
   563→                                iteration: context.iteration,
   564→                                tokensBefore: requestInputTokens,
   565→                                tokensAfter: afterRequestTokens,
   566→                                messagesBefore: beforeMessageCount,
   567→                                messagesAfter: result.messages.length,
   568→                                maxInputTokens,
   569→                        });
   570→                        captureCompactionExecuted(config.telemetry, {
   571→                                ulid: telemetryUlid,
   572→                                strategy: executedStrategy,
   573→                                mode: effectiveMode,
   574→                                messagesBefore: beforeMessageCount,
   575→                                messagesAfter: result.messages.length,
   576→                                messagesRemoved: beforeMessageCount - result.messages.length,
   577→                                tokensBefore: requestInputTokens,
   578→                                tokensAfter: afterRequestTokens,
   579→                                tokensSaved: requestInputTokens - afterRequestTokens,
   580→                                triggerTokens: requestTriggerTokens,
   581→                                maxInputTokens,
   582→                                thresholdRatio: COMPACTION_TRIGGER_RATIO,
   583→                                durationMs,
   584→                                // Matches the field name used by other TASK telemetry helpers
   585→                                // (e.g. captureTaskCompleted, captureToolUsage).
   586→                                provider: config.providerId,
   587→                                modelId: config.modelId,
   588→                                ...telemetryIdentity,
   589→                        });
   590→                        if (
   591→                                result.budget &&
   592→                                (result.budget.actionCount > 0 || result.budget.warningCount > 0)
   593→                        ) {
   594→                                captureCompactionBudgetEmergency(config.telemetry, {
   595→                                        ulid: telemetryUlid,
   596→                                        strategy: executedStrategy,
   597→                                        mode: effectiveMode,
   598→                                        policyIntent: result.budget.policyIntent,
   599→                                        actionCount: result.budget.actionCount,
   600→                                        warningCount: result.budget.warningCount,
   601→                                        liveTailHandling: result.budget.liveTailHandling,
   602→                                        provider: config.providerId,
   603→                                        modelId: config.modelId,
   604→                                        ...telemetryIdentity,
   605→                                });
   606→                                context.emitStatusNotice?.("compaction-budget-adjusted", {
   607→                                        kind: "compaction_budget_emergency",
   608→                                        reason: "compaction_budget_emergency",
   609→                                        iteration: context.iteration,
   610→                                        policyIntent: result.budget.policyIntent,
   611→                                        actionCount: result.budget.actionCount,
   612→                                        warningCount: result.budget.warningCount,
   613→                                });
   614→                        }
   615→                } else {
   616→                        context.emitStatusNotice?.(`${noticePrefix}compaction-skipped`, {
   617→                                kind: statusReason,
   618→                                reason: statusReason,
   619→                                phase: "skipped",
   620→                                iteration: context.iteration,
   621→                                maxInputTokens,
   622→                        });
   623→                        captureCompactionSkipped(config.telemetry, {
   624→                                ulid: telemetryUlid,
   625→                                strategy: executedStrategy,
   626→                                mode: effectiveMode,
   627→                                reason: "no_result",
   628→                                tokensBefore: requestInputTokens,
   629→                                triggerTokens: requestTriggerTokens,
   630→                                maxInputTokens,
   631→                                thresholdRatio: COMPACTION_TRIGGER_RATIO,
   632→                                durationMs,
   633→                                provider: config.providerId,
   634→                                modelId: config.modelId,
   635→                                ...telemetryIdentity,
   636→                        });
   637→                }
   638→
   639→                return result;
   640→        };
   641→}
   642→
   643→export function createCompactionStateAwarePrepareTurn(input: {
   644→        compact?: ContextPipelinePrepareTurn;
   645→        getState?: () => SessionCompactionState | undefined;
   646→        /**
   647→         * Persist a freshly-computed compaction state. `sourceMessages` are the
   648→         * exact canonical messages the state's source-prefix hash was computed
   649→         * over; hosts must validate projection against these rather than a
   650→         * separately derived transcript, which can legally differ mid-turn and
   651→         * spuriously reject the write.
   652→         */
   653→        saveState?: (
   654→                state: SessionCompactionState,
   655→                sourceMessages: CoreCompactionContext["messages"],
   656→        ) => void | Promise<void>;
   657→}): ContextPipelinePrepareTurn {
   658→        return async (context) => {
   659→                const existingState = input.getState?.();
   660→                const projectedMessages = existingState
   661→                        ? projectSessionCompactionState(existingState, context.messages)
   662→                        : undefined;
   663→                if (existingState && projectedMessages) {
   664→                        // Re-compaction intentionally starts from the compacted projection plus
   665→                        // canonical tail. This keeps automatic turns bounded without rebuilding a
   666→                        // full-transcript summary every turn; manual `/compact` is the path for a
   667→                        // fresh summary from canonical history.
   668→                        const result = input.compact
   669→                                ? await input.compact({
   670→                                                ...context,
   671→                                                messages: projectedMessages,
   672→                                                apiMessages: projectedMessages,
   673→                                        })
   674→                                : undefined;
   675→                        if (result?.messages) {
   676→                                const systemPrompt = result.systemPrompt ?? existingState.system_prompt;
   677→                                const nextState = createSessionCompactionState({
   678→                                        sourceMessages: context.messages,
   679→                                        compactedMessages: result.messages,
   680→                                        conversationId: context.conversationId,
   681→                                        systemPrompt,
   682→                                });
   683→                                await input.saveState?.(nextState, context.messages);
   684→                                return {
   685→                                        ...result,
   686→                                        ...(systemPrompt !== undefined ? { systemPrompt } : {}),
   687→                                };
   688→                        }
   689→                        return {
   690→                                messages: projectedMessages,
   691→                                ...(result?.systemPrompt !== undefined
   692→                                        ? { systemPrompt: result.systemPrompt }
   693→                                        : existingState.system_prompt !== undefined
   694→                                                ? { systemPrompt: existingState.system_prompt }
   695→                                                : {}),
   696→                        };
   697→                }
   698→                const result = input.compact ? await input.compact(context) : undefined;
   699→                if (result?.messages) {
   700→                        const nextState = createSessionCompactionState({
   701→                                sourceMessages: context.messages,
   702→                                compactedMessages: result.messages,
   703→                                conversationId: context.conversationId,
   704→                                systemPrompt: result.systemPrompt,
   705→                        });
   706→                        await input.saveState?.(nextState, context.messages);
   707→                }
   708→                return result;
   709→        };
   710→}
