     3→# Build (and optionally publish) the combined A/B VSIX: a tiny loader plus two
     4→# complete extension bundles — `next/` from the SDK-based apps/vscode on main,
     5→# `legacy/` from the legacy-extension branch. Cohort selection happens at
     6→# runtime via PostHog flags; see apps/vscode-rollout/README.md for the design
     7→# and the rollout runbook.
     8→#
     9→# Job layout: cheap input gates (preflight) and the two bundle test suites run
    10→# ungated; the build job packages the VSIX with no environment attached, so
    11→# publish=false rehearsals complete without any approval; only the publish job
    12→# — Marketplace + Open VSX + bookkeeping — waits on the `publish` environment.
    13→

    28→                required: true
    29→                default: "legacy-extension"
    30→                type: string
    31→            publish:
    32→                description: "Publish to the VS Code Marketplace and Open VSX (unchecked: just build the .vsix artifact)"
    33→                required: true
    34→                default: false
    35→                type: boolean
    36→

    38→    contents: read

    46→    # suites, the ~20-minute build, and the environment approval — instead of
    47→    # at publish time.
    48→    preflight:
    49→        name: Validate inputs
    50→        runs-on: ubuntu-latest
    51→        steps:
    52→            # The input reaches the shell ONLY via env here (never inline
    53→            # expression interpolation, which is evaluated before bash runs and
    54→            # would allow script injection from the dispatch form). Because
    55→            # every later job `needs` preflight, passing this regex is what
    56→            # makes the plain-string `${{ inputs.version }}` interpolations
    57→            # downstream safe.
    58→            - name: Validate version format
    59→              env:
    60→                  VERSION: ${{ github.event.inputs.version }}
    61→              run: |
    62→                  if [[ ! "$VERSION" =~ ^[0-9]+\.[0-9]+\.[0-9]+$ ]]; then
    63→                    echo "Error: version must be plain X.Y.Z with no leading 'v' and no suffix (got '$VERSION')."
    64→                    echo "It is stamped verbatim into the union manifest and both bundle manifests."
    65→                    exit 1
    66→                  fi
    67→                  echo "Version format ok: $VERSION"
    68→
    69→            # The reusable bun suite tests the dispatch revision (main), so
    70→            # publishing any other next-ref would ship an untested bundle.
    71→            # Build-only runs (publish=false) may still use arbitrary next-refs
    72→            # for artifact rehearsals.
    73→            - name: Refuse to publish an untested next-ref
    74→              if: ${{ github.event.inputs.publish == 'true' && github.event.inputs.next-ref != 'main' }}
    75→              run: |
    76→                  echo "Error: publish=true requires next-ref=main — the test gate only covers main."
    77→                  exit 1
    78→
    79→            # Marketplace versions are monotonic and cannot be unpublished:
    80→            # every publish must exceed the highest version ever published to
    81→            # the claude-dev listing FROM ANY BRANCH (combined stable or legacy
    82→            # hotfix). The publish job re-checks right before publishing — the
    83→            # environment-approval wait can last days and a legacy hotfix can
    84→            # land in between. Keep both copies of this check in sync.
    85→            - name: Verify version exceeds the live Marketplace version
    86→              if: ${{ github.event.inputs.publish == 'true' }}
    87→              env:
    88→                  VERSION: ${{ github.event.inputs.version }}
    89→              run: |
    90→                  LIVE=$(curl -sf --retry 3 -X POST "https://marketplace.visualstudio.com/_apis/public/gallery/extensionquery" \
    91→                    -H "Content-Type: application/json" -H "Accept: application/json;api-version=3.0-preview.1" \
    92→                    --data '{"filters":[{"criteria":[{"filterType":7,"value":"saoudrizwan.claude-dev"}]}],"flags":16}' \
    93→                    | node -e 'let d="";process.stdin.on("data",c=>d+=c);process.stdin.on("end",()=>{process.stdout.write(JSON.parse(d).results[0].extensions[0].versions[0].version)})')
    94→                  if [[ -z "$LIVE" ]]; then
    95→                    echo "Error: could not resolve the live Marketplace version for saoudrizwan.claude-dev."
    96→                    exit 1
    97→                  fi
    98→                  node -e '
    99→                    const [next, live] = process.argv.slice(1).map((v) => v.split(".").map(Number));
   100→                    for (let i = 0; i < 3; i++) {
   101→                      if (next[i] > live[i]) process.exit(0);
   102→                      if (next[i] < live[i]) break;
   103→                    }
   104→                    console.error(`Error: version ${process.argv[1]} does not exceed the live Marketplace version ${process.argv[2]}.`);
   105→                    process.exit(1);
   106→                  ' "$VERSION" "$LIVE"
   107→                  echo "Version ok: $VERSION exceeds live Marketplace version $LIVE"
